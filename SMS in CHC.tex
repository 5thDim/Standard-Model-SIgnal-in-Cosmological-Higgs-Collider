\label{sec:gauge-boson}

\subsection{Preliminaries}
\label{sec:gauge-preliminaries}
We will compute the connected four-point function
$\langle\delta h_{\mb k_1}\delta h_{\mb k_2}
\delta h_{\mb k_3}\delta h_{\mb k_4}\rangle'$
generated by a massive gauge-boson loop. 
Our interest is the leading nonlocal signal in the collapsed $s$ channel,
\begin{equation}
    \mb k_s\equiv\mb k_1+\mb k_2=-(\mb k_3+\mb k_4),
    \qquad k_s\ll k_i,\qquad
    k_{12}\equiv k_1+k_2,\quad k_{34}\equiv k_3+k_4,
    \label{eq:collapsed-kinematics}
\end{equation}
where $k_i\equiv|\mb k_i|$. Thus $k_1\simeq k_2$ and
$k_3\simeq k_4$ at leading order. 
We choose $\mb k_s$ along the
$z$ axis and set the azimuth of $\mb k_1$ to zero. Here $\theta_1$ and
$\theta_3$ are the polar angles of $\mb k_1$ and $\mb k_3$, respectively, and
$\phi_3$ is their relative azimuth.

The Higgs--gauge couplings follow from the Higgs kinetic term
\cite{lu_cosmological_2020}:
\begin{equation}
    \frac{\sqrt{-g}}{2}|\mathrm D_\mu\mb H|^2
    \supset\frac{a^2g^2}{4}(h_0+\delta h)^2
    \left(W^+_\mu W^{-\mu}+\frac{Z_\mu Z^\mu}{2c_W^2}\right),
    \label{eq:higgs-gauge-couplings}
\end{equation}
where $g$ is the weak gauge coupling, $c_W=\cos\theta_W$, and the
Lorentz contractions on the right use the flat conformal metric.
The terms linear and quadratic in $\delta h$ generate the cubic and
quartic vertices, respectively. 
They give the bubble, two triangle placements, and box shown in
Fig.~\ref{fig:gauge-diagrams}.

In the light-Higgs regime, the dominant infrared Higgs loop contribution to the vector masses is
\cite{lu_cosmological_2020}
\begin{equation}
    m_W^2\simeq\frac{3g^2H^4}{8\pi^2m_H^2},\qquad
    m_Z^2\simeq\frac{m_W^2}{c_W^2}.
    \label{eq:gauge-masses}
\end{equation}
Here $m_H$ is the effective Higgs mass. We use a constant vector mass
$m_V$ and the parameter
$\tilde\nu\equiv\sqrt{m_V^2/H^2-1/4}>0$ in the following.
The $W$ and $Z$ contributions are obtained with their respective masses
and couplings in \eqref{eq:higgs-gauge-couplings}.

We use $\lambda,\sigma=L,+,-,T$ for vector polarization-basis labels, and
$\eta=\pm1$ for the two nonlocal branches.
The SK branch at vertex $v$ is denoted by $\aa_v=\pm1$, with $v=1,2,\ldots$.
Neglecting the Higgs mass in the external mode functions, 
the bulk-to-boundary propagator is:
\begin{equation}
    G_{\aa_v}(k;\tau)=\frac{H^2}{2k^3}
    (1-\ii\aa_v k\tau)e^{\ii\aa_v k\tau}.
    \label{eq:higgs-propagator}
\end{equation}
The gauge boson propagator decomposes as
\begin{align}
    D_{\aa_1\aa_2,\mu\nu}(\mb k;\tau_1,\tau_2)
    ={}&\sum_{\lambda=T,L,\pm}
    e^{(\lambda)}_{\mb k,\mu}e^{(\lambda)*}_{\mb k,\nu}
    D^{(\lambda)}_{\aa_1\aa_2}(k;\tau_1,\tau_2)
    \nonumber\\
    &+e^{(T)}_{\mb k,\mu}e^{(L)*}_{\mb k,\nu}
    D^{(TL)}_{\aa_1\aa_2}(k;\tau_1,\tau_2)
    +e^{(L)}_{\mb k,\mu}e^{(T)*}_{\mb k,\nu}
    D^{(LT)}_{\aa_1\aa_2}(k;\tau_1,\tau_2),
    \label{eq:polarization-decomposition}
\end{align}
where $\mu,\nu=0,1,2,3$. In the comoving frame, the covariant
polarization basis is \cite{qin_helical_2023}
\[
    e^{(T)}_{\mb k,\mu}=(1,\mb0),\qquad
    e^{(L)}_{\mb k,\mu}=(0,\widehat{\mb k}),\qquad
    e^{(\pm)}_{\mb k,\mu}
    =\frac1{\sqrt2}(0,\mb e_\theta\pm\ii\mb e_\phi),
\]
with $(\widehat{\mb k},\mb e_\theta,\mb e_\phi)$ the orthonormal
spherical basis. 
The label $T$ denotes the temporal component of the longitudinal mode.
At zero chemical potential, all components of the vector propagator can be expressed through an auxiliary scalar
propagator with the same mass index
\cite{qin_helical_2023,qin_closed-form_2023}:
\begin{align}
    D^{(\mathrm{s})}_>(k;\tau_1,\tau_2)
    &=\frac{\pi H^2}{4}e^{-\pi\tilde\nu}
    (\tau_1\tau_2)^{3/2}
    \mathrm H^{(1)}_{\ii\tilde\nu}(-k\tau_1)
    \mathrm H^{(2)}_{-\ii\tilde\nu}(-k\tau_2),
    \label{eq:scalar-propagator}\\
    D^{(\pm)}_>&=\frac{D^{(\mathrm{s})}_>}{H^2\tau_1\tau_2},\qquad
    D^{(L)}_>=\frac{(\partial_{\tau_1}-2/\tau_1)
    (\partial_{\tau_2}-2/\tau_2)D^{(\mathrm{s})}_>}
    {H^2(\tilde\nu^2+1/4)},
    \label{eq:vector-scalar-relation}\\
    D^{(T)}_>&=\frac{k^2D^{(\mathrm{s})}_>}
    {H^2(\tilde\nu^2+1/4)},
    \label{eq:temporal-scalar-relation}\\
    D^{(TL)}_>&=\frac{\ii k(\partial_{\tau_2}-2/\tau_2)D^{(\mathrm{s})}_>}
    {H^2(\tilde\nu^2+1/4)},\qquad
    D^{(LT)}_>=-\frac{\ii k(\partial_{\tau_1}-2/\tau_1)D^{(\mathrm{s})}_>}
    {H^2(\tilde\nu^2+1/4)}.
    \label{eq:mixed-scalar-relations}
\end{align}

The Schwinger-Keldysh propagators can be constructed from the greater propagator.
We have $D_{-+}=D_>$, $D_{+-}=D_<$, and the $++$ and $--$ propagators are
\begin{align}
    D_{++}&=\theta(\tau_1-\tau_2)D_>
              +\theta(\tau_2-\tau_1)D_<,\nonumber\\
    D_{--}&=\theta(\tau_2-\tau_1)D_>
              +\theta(\tau_1-\tau_2)D_<.
    \label{eq:sk-prescription}
\end{align}
These relations apply to $D^{(\pm)},D^{(L)},D^{(TL)}$ and $D^{(LT)}$. 
But for $D^{(T)}$, the $++$ and $--$ propagators have a delta-function term because the temporal mode is not a physical degree of freedom:
\begin{align}
    D_{++}^{(T)}&=\theta(\tau_1-\tau_2)D_>^{(T)}
              +\theta(\tau_2-\tau_1)D_<^{(T)} + \frac{\ii \delta(\tau_1-\tau_2)}{H^2(\tn^2+1/4) a(\tau_1)^2} ,\nonumber\\
    D_{--}^{(T)}&=\theta(\tau_2-\tau_1)D_>^{(T)}
              +\theta(\tau_1-\tau_2)D_<^{(T)} - \frac{\ii \delta(\tau_1-\tau_2)}{H^2(\tn^2+1/4) a(\tau_1)^2}.
\end{align}
We will show that the propagators $D^{(T)},D^{(TL)}$ and $D^{(LT)}$ do not contribute to the leading nonlocal signal
under the one-loop factorization theorem.

The leading nonlocal parts of the spatial propagators are independent
of the SK indices:
\begin{align}
    \left[D^{(\pm)}_{\aa_1\aa_2}(k;\tau_1,\tau_2)\right]_{\mathrm{NonL}}
    &=\frac{(\tau_1\tau_2)^{1/2}}{4\pi}
      \sum_{\eta=\pm1}\Gamma^2(-\ii\eta\tilde\nu)
      \left(\frac{k^2\tau_1\tau_2}{4}\right)^{\ii\eta\tilde\nu},
      \label{eq:transverse-nonlocal}\\
    \left[D^{(L)}_{\aa_1\aa_2}(k;\tau_1,\tau_2)\right]_{\mathrm{NonL}}
    &=\frac{(\tau_1\tau_2)^{1/2}}{4\pi}
      \sum_{\eta=\pm1}
      \frac{(-1/2+\ii\eta\tilde\nu)^2}{\tilde\nu^2+1/4}
      \Gamma^2(-\ii\eta\tilde\nu)
      \left(\frac{k^2\tau_1\tau_2}{4}\right)^{\ii\eta\tilde\nu}.
    \label{eq:longitudinal-nonlocal}
\end{align}
The temporal nonlocal part is
\begin{equation}
    \left[D^{(T)}_{\aa_1\aa_2}(k;\tau_1,\tau_2)\right]_{\mathrm{NonL}}
    =\frac{k^2(\tau_1\tau_2)^{3/2}}
    {4\pi(\tilde\nu^2+1/4)}
    \sum_{\eta=\pm1}\Gamma^2(-\ii\eta\tilde\nu)
    \left(\frac{k^2\tau_1\tau_2}{4}\right)^{\ii\eta\tilde\nu}.
    \label{eq:temporal-nonlocal}
\end{equation}

The one-loop factorization theorem and cutting rule
\cite{qin_inflation_2023} isolate the leading nonlocal signal in the
collapsed channel. 
We first identify the two internal lines,
whose momenta can be chosen as $\mb q$ and
$\mb K\equiv\mb q+\mb k_s$. The relevant loop region has
$q,K\sim k_s\ll k_i$. We replace these two propagators by the
leading nonlocal terms of their late-time expansions, keeping their
polarization tensors, and expand the remaining hard propagators
to leading order in the soft momenta. The hard propagators retain
their full time dependence and SK ordering.
The cut propagators are independent of the SK indices, and their
time dependence separates between the two endpoints. The time
integrals thus factorize into two tree subgraphs with additional
time powers supplied by the cut lines. Multiplying these subgraphs
by the soft momentum integral and summing over polarizations gives
the nonlocal signal. 
The branches $\eta=\pm1$ combine into the oscillatory signal
retained below. Higher terms in the soft expansion give subleading
corrections in the collapsed limit.

Now we will show that the propagators $D^{(T)}$, $D^{(TL)}$, and $D^{(LT)}$ do not
contribute to the leading nonlocal signal. On a soft line, each
temporal mode function $B^{(T)}$ carries an additional factor of
$k\tau$ relative to the spatial modes $B^{(L)}$ and $B^{(\pm)}$.
Their nonanalytic contributions are therefore suppressed in the
collapsed limit. The contact term in $D^{(T)}$ is analytic in the
soft momentum and does not contribute to the nonlocal signal.
On a hard line, a temporal endpoint has a vanishing polarization
contraction with the spatial components of the attached soft line.
A nonzero contraction would require a temporal soft component,
which is already suppressed. We therefore omit propagators with
temporal components when computing the leading signal below.

All required soft momentum integrals reduce to five dimensionless
functions $f_r$, $r=0,1,2,3,4$. With
$z\equiv\widehat{\mb q}\cdot\widehat{\mb k}_s$, we define
\begin{equation}
    \left[ \int\frac{\di^3\mb q}{(2\pi)^3}
    q^{-2\alpha}K^{-2\beta}z^r \right]_{\mathsf{NonL}}
    =\frac{k_s^{3-2\alpha-2\beta}}{(4\pi)^{3/2}}
    f_r(\alpha,\beta),\qquad r=0,1,2,3,4.
    \label{eq:fr-definition}
\end{equation}
These expressions are understood by analytic continuation from the
convergent domain, with only the nonanalytic term retained.
The first three functions are
\begin{align}
    f_0(\alpha,\beta)
    &=\frac{\Gamma(\alpha+\beta-3/2)
    \Gamma(3/2-\alpha)\Gamma(3/2-\beta)}
    {\Gamma(3-\alpha-\beta)\Gamma(\alpha)\Gamma(\beta)},
    \label{eq:f0}\\
    f_1(\alpha,\beta)
    &=\frac12\left[
    f_0(\alpha+\tfrac12,\beta-1)
    -f_0(\alpha-\tfrac12,\beta)
    -f_0(\alpha+\tfrac12,\beta)\right],
    \label{eq:f1}\\
    f_2(\alpha,\beta)
    &=\frac14\left[
    f_0(\alpha+1,\beta-2)+f_0(\alpha-1,\beta)
    +f_0(\alpha+1,\beta)\right.\nonumber\\
    &\hspace{13mm}\left.
    -2f_0(\alpha,\beta-1)-2f_0(\alpha+1,\beta-1)
    +2f_0(\alpha,\beta)\right].
    \label{eq:f2}
\end{align}
The functions $f_3$ and $f_4$ are likewise finite sums of $f_0$
and their explicit expressions are collected in App.~\ref{app:loop-integrals}.

For the nonlocal branch $\eta$, it is convenient to set
$x=-\ii\eta\tilde\nu$ and define
\begin{equation}
    \mathcal J_r(m,n;x)
    \equiv f_r\left(x-\frac m2,x+\frac n2\right),
    \qquad r=0,1,2,3,4.
    \label{eq:Jr-definition}
\end{equation}
Here $m,n$ are integer shifts of momentum powers.
In particular,
\begin{equation}
    \int\frac{\di^3\mb q}{(2\pi)^3}
    \left(\frac q2\right)^{-2x}
    \left(\frac K2\right)^{-2x}q^mK^{-n}z^r
    =\frac{4^{2x}k_s^{3-4x+m-n}}{(4\pi)^{3/2}}
    \mathcal J_r(m,n;x).
    \label{eq:Jr-normalization}
\end{equation}

\subsection{Bubble Diagram}
\label{sec:gauge-bubble}
The bubble contains two quartic vertices, each attached to a pair of external Higgs modes. 
With the four external momenta fixed, the $s$-channel contribution is
\begin{align}
    \langle\delta h^4\rangle'_{\mathrm{Bub}}
    ={}&\frac{g^4}{8}
    \sum_{\aa_1,\aa_2=\pm1}(-\aa_1\aa_2)
    \int_{-\infty}^{0}\frac{\di\tau_1}{|H\tau_1|^2}
    \int_{-\infty}^{0}\frac{\di\tau_2}{|H\tau_2|^2}
    G_{\aa_1}(k_1;\tau_1)G_{\aa_1}(k_2;\tau_1)
    G_{\aa_2}(k_3;\tau_2)G_{\aa_2}(k_4;\tau_2)\nonumber\\
    &\times\int\frac{\di^3\mb q}{(2\pi)^3}
    D_{\aa_1\aa_2,ij}(\mb q;\tau_1,\tau_2)
    D_{\aa_2\aa_1,ji}(\mb K;\tau_2,\tau_1),
    \qquad \mb K=\mb q+\mb k_s.
    \label{eq:bubble-sk}
\end{align}
The symmetry factor $1/2$ which is associated with exchanging the two identical
internal lines is included. For the fixed partition $(12)\mid(34)$,
exchanges within either external Higgs pair leave the graph unchanged.
There is therefore no additional external-permutation factor in the
$s$-channel signal. 
Throughout this section,
the $W^\pm$ contribution is obtained by multiplying the expressions
by $2$, and the $Z$ contribution by making the replacement $g\to g/c_W$,
with the corresponding mass used in each case.

Replacing the two soft propagators by
\eqref{eq:transverse-nonlocal}--\eqref{eq:longitudinal-nonlocal}
removes their SK-index dependence, as illustrated in
Fig.~\ref{fig:bubble-cutting}.
The oscillatory signal comes from the two soft propagators
with the same branch label $\eta$.
After summing the polarizations, this contribution is
\begin{align}
    &\left[\int\frac{\di^3\mb q}{(2\pi)^3}
    D_{\aa_1\aa_2,ij}(\mb q;\tau_1,\tau_2)
    D_{\aa_2\aa_1,ji}(\mb K;\tau_2,\tau_1)\right]_{\mathrm{NonL}}
    \nonumber\\
    &\qquad\simeq\sum_{\eta=\pm1}
    (\tau_1\tau_2)^{1+2\ii\eta\tilde\nu}
    \frac{\Gamma^4(-\ii\eta\tilde\nu)}{(4\pi)^{7/2}}
    4^{-2\ii\eta\tilde\nu}k_s^{3+4\ii\eta\tilde\nu}
    \mathcal I_{\mathrm{Bub}}(-\ii\eta\tilde\nu).
    \label{eq:bubble-cut-loop}
\end{align}
\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{figure/Bub NL.pdf}
    \caption{The leading nonlocal part of the bubble in the
    $k_s\to0$ limit. The two soft gauge-boson lines supply the
    central loop factor, while the left and right subgraphs give
    the two quartic-vertex time integrals. }
    \label{fig:bubble-cutting}
\end{figure}

The dimensionless polarization sum has the compact form
\begin{align}
    \mathcal I_{\mathrm{Bub}}(x)
    ={}&\frac{3+2x}{1-2x}\,\mathcal J_0(0,0;x)
    +\frac{16x^2}{(1-2x)^2}
    \left[\mathcal J_0(2,2;x)+2\mathcal J_1(1,2;x)
    +\mathcal J_2(0,2;x)\right],
    \label{eq:bubble-polarization-sum}
\end{align}
with $x=-\ii\eta\tilde\nu$ as in \eqref{eq:Jr-definition}.
The separate polarization contributions are given in
App.~\ref{app:bubble-polarizations}. 

The remaining time integrals factorize. For the left subgraph, at
$k_1\simeq k_2\simeq k_{12}/2$, we find
\begin{align}
    &\sum_{\aa_1=\pm1}\ii\aa_1
    \int_{-\infty}^{0}\frac{\di\tau}{|H\tau|^2}
    G_{\aa_1}(k_1;\tau)G_{\aa_1}(k_2;\tau)
    (-\tau)^{1+2\ii\eta\tilde\nu}
    \simeq\frac{H^2k_{12}^{-2\ii\eta\tilde\nu}}
    {4k_1^3k_2^3}\,\mathcal P^{(1)}_{\eta}(\tilde\nu),
    \label{eq:bubble-time-integral}\\
    &\mathcal P^{(1)}_{\eta}(\tilde\nu)
    =\frac{2+\ii\eta\tilde\nu}{2\ii\eta\tilde\nu}
    \ii\sinh(\pi\eta\tilde\nu)\Gamma(2+2\ii\eta\tilde\nu).
    \label{eq:bubble-P1}
\end{align}
The right vertex gives the same expression with
$(k_1,k_2,k_{12})\to(k_3,k_4,k_{34})$.
Combining the two vertices with \eqref{eq:bubble-cut-loop}, we obtain
\begin{equation}
    \left[\langle\delta h^4\rangle'_{\mathrm{Bub}}\right]_{\mathrm{NonL}}
    \simeq\frac{g^4H^4k_s^3}
    {128k_1^3k_2^3k_3^3k_4^3}
    \left[
    \left(\frac{k_s^2}{4k_{12}k_{34}}\right)^{2\ii\tilde\nu}
    C^{\mathrm{Bub}}(\tilde\nu)+\text{c.c.}\right],
    \label{eq:bubble-signal}
\end{equation}
where the complex signal coefficient is
\begin{equation}
    C^{\mathrm{Bub}}(\tilde\nu)
    =\frac{\Gamma^4(-\ii\tilde\nu)}{(4\pi)^{7/2}}
    \left[\mathcal P^{(1)}_{+}(\tilde\nu)\right]^2
    \mathcal I_{\mathrm{Bub}}(-\ii\tilde\nu).
    \label{eq:bubble-coefficient}
\end{equation}
This definition separates the loop and time integrals from the

    \label{eq:bubble-phase}
\end{equation}

\subsection{Triangle Diagram}
\label{sec:gauge-triangle}
The triangle diagram contains two cubic vertices and one quartic vertex.
We first consider the diagram ``Triangle 1''.
The triangle has a unit diagram symmetry factor. 
In the conventions of Sec.~\ref{sec:gauge-bubble}, its integral expression is
\begin{align}
    \langle\delta h^4\rangle'_{\mathrm{Tri1}}
    ={}& -\frac{g^6h_0^2}{8}
    \sum_{\aa_1,\aa_2,\aa_3=\pm1}(-\ii\aa_1\aa_2\aa_3)
    \int_{-\infty}^0\prod_{v=1}^{3}
    \frac{\di\tau_v}{|H\tau_v|^2}\,
    G_{\aa_1}(k_1;\tau_1)G_{\aa_2}(k_2;\tau_2)
    G_{\aa_3}(k_3;\tau_3)G_{\aa_3}(k_4;\tau_3)
    \nonumber\\
    &\times\int\frac{\di^3\mb q}{(2\pi)^3}
    D_{\aa_1\aa_2,ij}(\mb q+\mb k_1;\tau_1,\tau_2)
    D_{\aa_2\aa_3,j\ell}(\mb K;\tau_2,\tau_3)
    D_{\aa_3\aa_1,\ell i}(\mb q;\tau_3,\tau_1).
    \label{eq:triangle-sk}
\end{align}
The nonlocal signal comes from $q,K\sim k_s\ll k_1$.
We cut the $\mb q$ and $\mb K$ lines and evaluate the remaining hard
propagator at $\mb q+\mb k_1\simeq\mb k_1$. 
The resulting factorization is shown in Fig.~\ref{fig:triangle-cutting}.

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{figure/Tri NL.pdf}
    \caption{The leading nonlocal part of the first triangle.
    Cutting the two soft lines leaves a hard gauge-boson exchange
    on the left and a quartic vertex on the right, joined by the
    central soft loop factor. The second triangle follows by
    exchanging the two external pairs.}
    \label{fig:triangle-cutting}
\end{figure}

After summing the soft polarizations, the cut loop takes the form
\begin{align}
    &\left[\int\frac{\di^3\mb q}{(2\pi)^3}
    D_{\aa_1\aa_2,ij}(\mb q+\mb k_1;\tau_1,\tau_2)
    D_{\aa_2\aa_3,j\ell}(\mb K;\tau_2,\tau_3)
    D_{\aa_3\aa_1,\ell i}(\mb q;\tau_3,\tau_1)\right]_{\mathrm{NonL}}
    \nonumber\\
    &\quad\simeq\sum_{\lambda=L,\pm}\sum_{\eta=\pm1}
    D^{(\lambda)}_{\aa_1\aa_2}(k_1;\tau_1,\tau_2)
    (\tau_1\tau_2)^{1/2+\ii\eta\tilde\nu}
    (-\tau_3)^{1+2\ii\eta\tilde\nu}
    \frac{\Gamma^4(-\ii\eta\tilde\nu)}{(4\pi)^{7/2}}
    4^{-2\ii\eta\tilde\nu}k_s^{3+4\ii\eta\tilde\nu}
    \mathcal I^\lambda_{\mathrm{Tri}}(-\ii\eta\tilde\nu,\theta_1).
    \label{eq:triangle-cut-loop}
\end{align}
This equation defines the dimensionless coefficients
$\mathcal I^\lambda_{\mathrm{Tri}}(x,\theta)$, with
$x=-\ii\eta\tilde\nu$. 
% Rotational symmetry and polarization
% completeness imply
% \begin{align}
%     \mathcal I^L_{\mathrm{Tri}}(x,\theta)
%     &=\sin^2\theta\,\mathcal I^L_{\mathrm{Tri}}(x,\pi/2)
%       +\cos^2\theta\,\mathcal I^L_{\mathrm{Tri}}(x,0),
%       \label{eq:triangle-angular-structure}\\
%     \mathcal I^+_{\mathrm{Tri}}(x,\theta)
%     &=\mathcal I^-_{\mathrm{Tri}}(x,\theta)
%       =\frac12\left[\mathcal I_{\mathrm{Bub}}(x)
%       -\mathcal I^L_{\mathrm{Tri}}(x,\theta)\right].
%       \label{eq:triangle-completeness}
% \end{align}
% Thus only one longitudinal coefficient is needed in addition to the bubble sum. 
% Its expression in terms of $\mathcal J_r$ is given in App.~\ref{app:triangle-coefficients}. 
After simplification, we find that all three
coefficients equal $\mathcal I_{\mathrm{Bub}}(x)/3$ and the apparent angular
dependence cancels, as shown in \eqref{eq:triangle-I-closed}.

We next evaluate the left and right subgraphs.
The quartic vertex gives the time integral
$\mathcal P^{(1)}_{\eta}$ in \eqref{eq:bubble-P1}.
The two cubic vertices and the hard line instead define
\begin{align}
    &\sum_{\aa_1,\aa_2=\pm1}(-\aa_1\aa_2)
    \int_{-\infty}^0\frac{\di\tau_1\di\tau_2}
    {|H\tau_1|^2|H\tau_2|^2}
    G_{\aa_1}(k_1;\tau_1)G_{\aa_2}(k_2;\tau_2)
    D^{(\lambda)}_{\aa_1\aa_2}(k_1;\tau_1,\tau_2)
    (\tau_1\tau_2)^{1/2+\ii\eta\tilde\nu}
    \nonumber\\
    &\qquad\simeq
    \frac{k_{12}^{-2\ii\eta\tilde\nu}}{4k_1^3k_2^3}
    \mathcal P^{(2),\lambda}_{\eta}(\tilde\nu).
    \label{eq:triangle-time-integral}
\end{align}
At leading order, $k_1\simeq k_2\simeq k_{12}/2$.
The integral therefore reduces to the
double-folded scalar seeds of Ref.~\cite{qin_closed-form_2023},
which we denote by
$\mathcal M_{\aa_1\aa_2}^{p_1p_2}\equiv
\widetilde{\mathcal I}_{\aa_1\aa_2}^{p_1p_2}(1,1)$.
For the transverse modes this follows directly from
\eqref{eq:vector-scalar-relation}.
For the longitudinal mode, the differential relation also produces
a contact contribution by integration by parts, and we retain this term together with the scalar-seed contributions.
The required seed exponents include
$p_0=-5/2+\ii\eta\tilde\nu$, on the boundary of the convergence
domain. We define the result by analytic continuation from
$p_0+\epsilon$, $\epsilon>0$, and take the finite part of the assembled
expression at $\epsilon=0$. The seed normalization, closed forms,
and combinations giving $\mathcal P^{(2),\lambda}_{\eta}$ are collected
in App.~\ref{app:triangle-coefficients}.

Finally, the leading signal of the first triangle is
\begin{equation}
    \left[\langle\delta h^4\rangle'_{\mathrm{Tri1}}\right]_{\mathrm{NonL}}
    \simeq -\frac{g^6h_0^2H^2k_s^3}
    {128k_1^3k_2^3k_3^3k_4^3}
    \left[
    \left(\frac{k_s^2}{4k_{12}k_{34}}\right)^{2\ii\tilde\nu}
    C^{\mathrm{Tri}}(\tilde\nu)+\text{c.c.}\right],
    \label{eq:triangle-single-signal}
\end{equation}
where
\begin{equation}
    C^{\mathrm{Tri}}(\tilde\nu)
    =\frac{\Gamma^4(-\ii\tilde\nu)}{(4\pi)^{7/2}}
    \mathcal P^{(1)}_{+}(\tilde\nu)
    \sum_{\lambda=L,\pm}
    \mathcal P^{(2),\lambda}_{+}(\tilde\nu)
    \mathcal I^\lambda_{\mathrm{Tri}}(-\ii\tilde\nu,\theta).
    \label{eq:triangle-coefficient}
\end{equation}
Since $\mathcal P^{(2),+}_{\eta}=\mathcal P^{(2),-}_{\eta}$, the two
transverse helicities contribute equally. 
The simplified loop
coefficients make $C^{\mathrm{Tri}}$ independent of $\theta$ at this order
and its reduced expression is given in \eqref{eq:triangle-C-reduced}.

In the collapsed $s$ channel, the quartic vertex carries either the
pair $(3,4)$, as in Triangle 1, or the pair $(1,2)$, as in Triangle 2.
For either placement, exchanging the two Higgs legs at the cubic
vertices reproduces the same undirected triangle after reflection.
Exchanging the two legs at the quartic vertex also leaves the graph
unchanged, since both attach to the same vertex. Each placement is
therefore counted once. Triangle 2 follows from Triangle 1 by
$(1,2)\leftrightarrow(3,4)$, and their leading collapsed signals are
equal. Adding these two contributions gives
\begin{align}
    \left[\langle\delta h^4\rangle'_{\mathrm{Tri}}\right]_{\mathrm{NonL}}
    \simeq{}&-\frac{g^6h_0^2H^2k_s^3}
    {64k_1^3k_2^3k_3^3k_4^3}
    \left\{\left(\frac{k_s^2}{4k_{12}k_{34}}\right)^{2\ii\tilde\nu}
    C^{\mathrm{Tri}}(\tilde\nu)
    +\text{c.c.}\right\}.
    \label{eq:triangle-total-signal}
\end{align}
The nonanalytic momentum dependence is the same as in the bubble,
while the hard propagator changes the amplitude and phase.
Figure~\ref{fig:triangle-coefficient} shows the coefficient
of a single triangle.

\subsection{Box Diagram}
\label{sec:gauge-box}
We first consider the box with cyclic ordering $1234$, shown in
Fig.~\ref{fig:box-cutting}. With the external labels fixed, its
diagram symmetry factor is unity. The four cubic vertices give the
coupling factor $(g^2h_0/2)^4=g^8h_0^4/16$.
In the conventions of Sec.~\ref{sec:gauge-bubble}, its $s$-channel
contribution is
\begin{align}
    &\langle\delta h^4\rangle'_{\mathrm{Box},1234}
    =\frac{g^8h_0^4}{16}
    \sum_{\aa_1,\aa_2,\aa_3,\aa_4=\pm1}\aa_1\aa_2\aa_3\aa_4
    \int_{-\infty}^0\prod_{v=1}^{4}
    \frac{\di\tau_v}{|H\tau_v|^2}
    \nonumber\\
    &\qquad\times G_{\aa_1}(k_1;\tau_1)G_{\aa_2}(k_2;\tau_2)
    G_{\aa_3}(k_3;\tau_3)G_{\aa_4}(k_4;\tau_4)
    \nonumber\\
    &\qquad\times\int\frac{\di^3\mb q}{(2\pi)^3}
    D_{\aa_1\aa_2,ij}(\mb q+\mb k_1;\tau_1,\tau_2)
    D_{\aa_2\aa_3,j\ell}(\mb K;\tau_2,\tau_3)
    \nonumber\\
    &\qquad\times D_{\aa_3\aa_4,\ell m}(\mb q-\mb k_4;\tau_3,\tau_4)
    D_{\aa_4\aa_1,mi}(\mb q;\tau_4,\tau_1),
    \label{eq:box-sk}
\end{align}
where $\mb K=\mb q+\mb k_s$. 

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{figure/Box NL.pdf}
    \caption{The leading nonlocal part of the box with cyclic ordering $1234$.
    Cutting the two soft lines separates
    two tree subgraphs, each containing one hard gauge-boson
    propagator. The central factor contains the nonlocal parts
    of the two soft propagators and their loop momentum integral.}
    \label{fig:box-cutting}

In the collapsed limit, the leading nonlocal signal in
\eqref{eq:box-sk} comes from the region $q,K\sim k_s$.
Cutting these two soft lines leaves
two hard propagators, with momenta approaching $\mb k_1$ and
$\mb k_3$. 
Their polarizations are denoted by
$\lambda_1,\lambda_3=L,\pm$.
The soft loop gives the dimensionless coefficient
$\mathcal I^{\lambda_1\lambda_3}_{\mathrm{Box}}
(-\ii\eta\tilde\nu,\theta_1,\theta_3,\phi_3)$ defined in
\eqref{eq:box-I-definition}. 
Its common momentum factor is
$4^{-2\ii\eta\tilde\nu}k_s^{3+4\ii\eta\tilde\nu}$,
as for the bubble and triangle.
Explicitly, after summing the soft polarizations, the cut loop is
\begin{align}
    &\left[\int\frac{\di^3\mb q}{(2\pi)^3}
    D_{\aa_1\aa_2,ij}(\mb q+\mb k_1;\tau_1,\tau_2)
    D_{\aa_2\aa_3,j\ell}(\mb K;\tau_2,\tau_3)\times
    D_{\aa_3\aa_4,\ell m}(\mb q-\mb k_4;\tau_3,\tau_4)
    D_{\aa_4\aa_1,mi}(\mb q;\tau_4,\tau_1)\right]_{\mathrm{NonL}}
    \nonumber\\
    &\quad\simeq\sum_{\lambda_1,\lambda_3=L,\pm}\sum_{\eta=\pm1}
    D^{(\lambda_1)}_{\aa_1\aa_2}(k_1;\tau_1,\tau_2)
    D^{(\lambda_3)}_{\aa_3\aa_4}(k_3;\tau_3,\tau_4)
    (\tau_1\tau_2)^{1/2+\ii\eta\tilde\nu}
    (\tau_3\tau_4)^{1/2+\ii\eta\tilde\nu}
    \nonumber\\
    &\qquad\times
    \frac{\Gamma^4(-\ii\eta\tilde\nu)}{(4\pi)^{7/2}}
    4^{-2\ii\eta\tilde\nu}k_s^{3+4\ii\eta\tilde\nu}
    \mathcal I^{\lambda_1\lambda_3}_{\mathrm{Box}}
    (-\ii\eta\tilde\nu,\theta_1,\theta_3,\phi_3).
    \label{eq:box-cut-loop}
\end{align}

As shown in Fig.~\ref{fig:box-cutting}, the remaining time integrals
factorize into two copies of
\eqref{eq:triangle-time-integral}, one for each hard line.
The left and right subgraphs contribute
$\mathcal P^{(2),\lambda_1}_{\eta}(\tilde\nu)$ and
$\mathcal P^{(2),\lambda_3}_{\eta}(\tilde\nu)$, respectively.

The collapsed $s$ channel contains a second inequivalent box,
obtained from the diagram shown in Fig.~\ref{fig:box-cutting} by
exchanging the external momenta $\mb k_1$ and $\mb k_2$. Exchanging
$\mb k_3$ and $\mb k_4$ gives the same additional undirected graph,
while exchanging both pairs returns the original graph.
At leading collapsed order, $\mb k_2\simeq-\mb k_1$, so the first
exchange sends $(\theta_1,\phi_3)$ to $(\pi-\theta_1,\phi_3+\pi)$.
The coefficient $C^{\mathrm{Box}}$ is invariant under this
transformation, as follows from its angular decomposition below.
The two boxes therefore contribute equally.

For each box, the vertex factor $1/16$ and the normalization $1/4$
from each hard subgraph give $1/256$. Summing the two distinct boxes
supplies a further factor $2$, giving $1/128$. Including the
complex-conjugate soft branch, the total $s$-channel signal is
\begin{align}
    \left[\langle\delta h^4\rangle'_{\mathrm{Box}}
    \right]_{\mathrm{NonL}}
    \simeq{}&\frac{g^8h_0^4k_s^3}
    {128k_1^3k_2^3k_3^3k_4^3}
    \left[
    \left(\frac{k_s^2}{4k_{12}k_{34}}\right)^{2\ii\tilde\nu}
    \times
    C^{\mathrm{Box}}(\tilde\nu,\theta_1,\theta_3,\phi_3)
    +\text{c.c.}\right].
    \label{eq:box-single-signal}
\end{align}
Here $C^{\mathrm{Box}}$ denotes the coefficient of the representative
box, before summing the external orderings:
\begin{align}
    C^{\mathrm{Box}}(\tilde\nu,\theta_1,\theta_3,\phi_3)
    ={}&\frac{\Gamma^4(-\ii\tilde\nu)}{(4\pi)^{7/2}}
    \sum_{\lambda_1,\lambda_3=L,\pm}
    \mathcal P^{(2),\lambda_1}_{+}(\tilde\nu)
    \mathcal P^{(2),\lambda_3}_{+}(\tilde\nu)
    \times\mathcal I^{\lambda_1\lambda_3}_{\mathrm{Box}}
    (-\ii\tilde\nu,\theta_1,\theta_3,\phi_3).
    \label{eq:box-coefficient}
\end{align}

Since the hard transverse propagators have equal time integrals,
the polarization sum reduces to the $LL$, $LH$, and $HH$ groups
in \eqref{eq:box-channel-groups} 
where $H$ means $+$ component contributions plus $-$ component contributions. 
Their explicit expressions are
given in App.~\ref{app:box-coefficients}. 
% In particular,
% \eqref{eq:box-C-reduced} shows that all angular dependence is
% carried by $\mathcal I^{LL}_{\mathrm{Box}}$ and multiplied by

% The difference between the longitudinal and transverse hard-line

    \nonumber\\
    &+B_g(\tilde\nu)\sin(2\theta_1)\sin(2\theta_3)\cos\phi_3
    \nonumber\\

    \label{eq:box-angular-expansion}
\end{align}
The five complex functions depend only on $\tilde\nu$.
The signal is symmetric
under $\theta_1\leftrightarrow\theta_3$ and
$\phi_3\to-\phi_3$. Its oscillatory momentum dependence is the
same as in \eqref{eq:bubble-phase}, with an angle-dependent
amplitude and phase supplied by $C^{\mathrm{Box}}$.

\section{Top Quark Signal}
\label{sec:top-quark}

\subsection{Preliminaries}
\label{sec:top-preliminaries}
We next compute the Higgs four-point function generated by the top-quark
loop. We use the collapsed kinematics of \eqref{eq:collapsed-kinematics}
and the external propagator \eqref{eq:higgs-propagator}. 
The Yukawa coupling and mass parameter are
\begin{equation}
 g_t\equiv\frac{y_t}{\sqrt2},\qquad m_t=g_th_0,\qquad
 \tilde\nu\equiv\frac{m_t}{H}>0.
 \label{eq:top-parameters}
\end{equation}
The relation between $\tilde\nu$ and the mass differs from that
of the vector field. 
% The constant-background approximation requires
% $|\dot h_0|\ll H|h_0|$, while the external Higgs modes are evaluated
% at $|m_H^2|\ll H^2$ \cite{lu_cosmological_2020}.

Let $t$ and $t^c$ denote the conformally rescaled left-handed top and
the left-handed charge conjugate of the right-handed top. Rescaling
the covariant fields by $a^{3/2}$, with $a=-1/(H\tau)$, gives
\begin{equation}
 \mathcal L=\ii t^\dagger\bar\sigma^\mu\partial_\mu t
 +\ii t^{c\dagger}\bar\sigma^\mu\partial_\mu t^c
 -a(m_t+g_t\delta h)(tt^c+\mathrm{h.c.}).
 \label{eq:top-dirac-action}
\end{equation}
Here $t^c$ represents the right-handed degree of freedom in left-handed
notation. Following the two-component description of
Ref.~\cite{aoki_fermionic_2026}, we set
$t=(\psi_1+\ii\psi_2)/\sqrt2$ and
$t^c=(\psi_1-\ii\psi_2)/\sqrt2$. The action becomes
\begin{equation}
 \mathcal L=\sum_{I=1}^2\left[
 \ii\psi_I^\dagger\bar\sigma^\mu\partial_\mu\psi_I
 -\frac a2(m_t+g_t\delta h)(\psi_I\psi_I+\mathrm{h.c.})\right].
 \label{eq:top-majorana-action}
\end{equation}
So now we have two identical spinor sectors.

We use $\sigma^a=(\one,\bm\sigma)$,
$\bar\sigma^a=(\one,-\bm\sigma)$, and
$\psi^\alpha=\epsilon^{\alpha\beta}\psi_\beta$ with
$\epsilon_{12}=1$, $\epsilon^{12}=-1$.
We write $\sigma_k=\bm\sigma\cdot\widehat{\mb k}$ and
$\epsmat=\ii\sigma^2=\bigl(\begin{smallmatrix}0&1\\-1&0\end{smallmatrix}\bigr)$. The helicity projectors are
$P_s=h^sh^{s\dagger}=(\one+s\sigma_k)/2$.
For each sector, the mode expansion is
\begin{equation}
 \psi(\tau,\mb x)=\int\frac{\di^3\mb k}{(2\pi)^3}
 \sum_{s=\pm1}\left[
 h^su_s b^s_{\mb k}e^{\ii\mb k\cdot\mb x}
 +\epsmat h^{s*}v_s^*b^{s\dagger}_{\mb k}e^{-\ii\mb k\cdot\mb x}
 \right],
 \label{eq:top-mode-expansion}
\end{equation}
where $\{b^s_{\mb k},b^{s'\dagger}_{\mb k'}\}
=(2\pi)^3\delta^{ss'}\delta^3(\mb k-\mb k')$.
The normalized Bunch--Davies state mode functions can be written as
\begin{equation}
 u_+=v_- =\frac{\tilde\nu}{\sqrt{2x}}
 W_{-1/2,\ii\tilde\nu}(-2\ii x),\qquad
 u_-=v_+ =\frac{\ii}{\sqrt{2x}}
 W_{1/2,\ii\tilde\nu}(-2\ii x),\qquad x=-k\tau.
 \label{eq:top-modes}
\end{equation}
The modes in \eqref{eq:top-modes} satisfy
$\ii\partial_\tau u_s+sku_s=am_tv_s$ and
$\ii\partial_\tau v_s-skv_s=am_tu_s$.
The helicity is $s=\pm1$, and the normalization gives $|u_s|^2+|v_s|^2=1$.
We use the following three types of propagators
\cite{you_neutrino_2026}
\begin{align}
 D_{\aa\bb,\alpha\dot\beta}(\mb k;\tau_1,\tau_2)
 &=\langle T_C\psi_{\aa,\alpha}(\mb k,\tau_1)
                      \psi^\dagger_{\bb,\dot\beta}(-\mb k,\tau_2)\rangle',
 \nonumber\\
 \widehat D_{\aa\bb,\alpha\beta}(\mb k;\tau_1,\tau_2)
 &=\langle T_C\psi_{\aa,\alpha}(\mb k,\tau_1)
                      \psi_{\bb,\beta}(-\mb k,\tau_2)\rangle',
 \nonumber\\
 \check D_{\aa\bb,\dot\alpha\dot\beta}(\mb k;\tau_1,\tau_2)
 &=\langle T_C\psi^\dagger_{\aa,\dot\alpha}(\mb k,\tau_1)
                      \psi^\dagger_{\bb,\dot\beta}(-\mb k,\tau_2)\rangle'.
 \label{eq:top-propagators}
\end{align}
Here $T_C$ denotes ordering along the Schwinger--Keldysh closed time
contour $C$. 
% For each of the three types, the time-ordered components obey
% \begin{equation}
%  D_{++}=\theta_{12} D_{-+}+\theta_{21} D_{+-},
%  \qquad
%  D_{--}=\theta_{21} D_{-+}+\theta_{12} D_{+-},
%  \qquad \theta_{ij}=\theta(\tau_i-\tau_j).
%  \label{eq:top-sk}
% \end{equation}
With $u_{sj}=u_s(k,\tau_j)$ and $v_{sj}=v_s(k,\tau_j)$,
the Wightman components of \eqref{eq:top-propagators} are
\begin{equation}
 D_{-+}=\sum_su_{s1}u_{s2}^*P_s,\qquad
 \widehat D_{-+}=-\sum_su_{s1}v_{s2}^*P_s\epsmat,\qquad
 \check D_{-+}=\sum_sv_{s1}u_{s2}^*\epsmat P_s.
 \label{eq:top-mode-contractions}
\end{equation}
The signs follow from
$h^s(\epsmat h^{s*})^T=-P_s\epsmat$.

For the seed reduction it is useful to express these propagators
through the scalar propagator with chemical potential
\cite{qin_closed-form_2023}. With $X=-k\tau_1$, $Y=-k\tau_2$,
\begin{align}
 \Dsc{\kappa}_{-+}(k;\tau_1,\tau_2)
 &=\frac{e^{\ii\pi\kappa}}{2k}
 W_{\kappa,\ii\tilde\nu}(-2\ii X)
 W_{-\kappa,\ii\tilde\nu}(2\ii Y),\nonumber\\
 \Dsc{\kappa}_{+-}(k;\tau_1,\tau_2)
 &=\frac{e^{\ii\pi\kappa}}{2k}
 W_{-\kappa,\ii\tilde\nu}(2\ii X)
 W_{\kappa,\ii\tilde\nu}(-2\ii Y).
 \label{eq:top-scalar-kernel}
\end{align}
Here $W_{\kappa,\ii\tilde\nu}$ is the Whittaker W function.
The time-ordered scalar components follow the prescription in
\eqref{eq:sk-prescription}.
In the notation of the reference, $\kappa=\ii h\widetilde\mu$,
and in the spinor case this parameter is analytically continued to
$\kappa=\pm1/2$.
We denote the kernels at these two values
explicitly by $\Dsc{+1/2}$ and $\Dsc{-1/2}$, respectively.

The three spinor propagators can then be written as
\begin{align}
 D_{\aa\bb}
 &=\frac{\ii k}{\sqrt{XY}}\left[
 (-X\partial_X-\ii X+\tfrac12)\Dsc{-1/2}_{\aa\bb}P_-
 -(X\partial_X-\ii X-\tfrac12)\Dsc{+1/2}_{\aa\bb}P_+\right],
 \nonumber\\
 \widehat D_{\aa\bb}
 &=-\frac{\tilde\nu k}{\sqrt{XY}}
 [\Dsc{-1/2}_{\aa\bb}P_++\Dsc{+1/2}_{\aa\bb}P_-]\epsmat,
 \nonumber\\
 \check D_{\aa\bb}
 &=\frac{\tilde\nu k}{\sqrt{XY}}\epsmat
 [\Dsc{+1/2}_{\aa\bb}P_++\Dsc{-1/2}_{\aa\bb}P_-].
 \label{eq:top-scalar-reconstruction}
\end{align}
The vertex-type notation and fermionic reversal rule are
\begin{equation}
 D^{\mathrm L\overline{\mathrm L}}=D,\quad
 D^{\mathrm L\mathrm L}=\widehat D,\quad
 D^{\overline{\mathrm L}\overline{\mathrm L}}=\check D,\quad
 D^{\overline{\mathrm L}\mathrm L}_{\aa\bb}(\mb k;\tau_1,\tau_2)
 =-[D_{\bb\aa}(-\mb k;\tau_2,\tau_1)]^T.
 \label{eq:top-reversal}
\end{equation}

The soft limit follows from the small-argument expansion
\begin{equation}
 W_{\kappa,\ii\tilde\nu}(z)
 =\sum_{\eta=\pm1}
 \frac{\Gamma(-2\ii\eta\tilde\nu)}
      {\Gamma(\tfrac12-\ii\eta\tilde\nu-\kappa)}
 z^{1/2+\ii\eta\tilde\nu}
 \left[1-\frac{\kappa z}{1+2\ii\eta\tilde\nu}+O(z^2)\right].
 \label{eq:top-whittaker-expansion}
\end{equation}

Substituting this expansion into the propagators gives, at
$|k\tau_1|,|k\tau_2|\ll1$, the required nonlocal terms
\begin{align}
 [D_{\aa\bb}]_{\mathrm{NonL}}
 &=-\frac1{4\pi}\sum_{\eta=\pm1}
 \Gamma^2(\tfrac12-\ii\eta\tilde\nu)
 \left(\frac{k^2\tau_1\tau_2}{4}\right)^{\ii\eta\tilde\nu}
 \left[\sigma_k+\frac{\ii k(\tau_1-\tau_2)}{1+2\ii\eta\tilde\nu}\one
 +\cdots\right],\nonumber\\
 [\widehat D_{\aa\bb}]_{\mathrm{NonL}}
 &=-\frac1{4\pi}\sum_{\eta=\pm1}\eta
 \Gamma^2(\tfrac12-\ii\eta\tilde\nu)
 \left(\frac{k^2\tau_1\tau_2}{4}\right)^{\ii\eta\tilde\nu}
 \left[\sigma_k+\frac{\ii k(\tau_1+\tau_2)}{1+2\ii\eta\tilde\nu}\one
 +\cdots\right]\epsmat,\nonumber\\
 [\check D_{\aa\bb}]_{\mathrm{NonL}}
 &=-\frac1{4\pi}\sum_{\eta=\pm1}\eta
 \Gamma^2(\tfrac12-\ii\eta\tilde\nu)
 \left(\frac{k^2\tau_1\tau_2}{4}\right)^{\ii\eta\tilde\nu}
 \epsmat\left[\sigma_k-\frac{\ii k(\tau_1+\tau_2)}{1+2\ii\eta\tilde\nu}\one
 +\cdots\right].
 \label{eq:top-soft-propagators}
\end{align}
The omitted terms are second order in the soft momentum at fixed times.
These nonlocal expressions are independent of the SK indices. They
allow the two soft lines to be cut, while the remaining hard lines
retain their full time dependence and SK ordering
\cite{qin_inflation_2023,you_neutrino_2026}.

In each sector, the Yukawa interaction gives two vertex types, labelled
$r=\mathrm L,\overline{\mathrm L}$ for $\psi\psi$ and
$\psi^\dagger\psi^\dagger$, respectively. 
Their amputated rules are
$-\ii\aa g_ta(\tau)J_r$, where
$J_{\mathrm L}=\epsmat$ and $J_{\overline{\mathrm L}}=-\epsmat$, and each interaction vertex retains one factor of $a(\tau)$.

\subsection{The Box Diagram and Soft Expansion}
\label{sec:top-box}
We first consider the box shown in Fig.~\ref{fig:top-box}, with ordering
$1234$, for a single spinor sector and a fixed color. We sum over all
assignments of the two Yukawa vertex types at its four vertices,
including the corresponding propagator combinations. 


\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.48\textwidth]{figure/quark box.pdf}
    \caption{The top-quark box with cyclic ordering $1234$.
    Black solid lines denote external Higgs modes and blue solid
    lines denote fermion propagators. The labels $r_i$ specify the
    Yukawa vertex types, which are summed independently at each vertex.}
    \label{fig:top-box}
\end{figure}

External-momentum
permutations will be included after the vertex-type sum. 
For the collapsed $s$-channel contribution, we retain the original term
together with those obtained by exchanging $\mb k_1\leftrightarrow\mb k_2$,
exchanging $\mb k_3\leftrightarrow\mb k_4$, and performing both exchanges.
The two single exchanges describe the same additional connection,
while the simultaneous exchange describes the original connection
traversed in reverse. 
Reversing the traversal does not generate new
Wick pairings, so this prescription represents each set of pairings
twice. 
We therefore multiply the four-term sum by $1/2$. 

With this normalization, the
sum over vertex types for ordering $1234$ is
\begin{align}
 \langle\delta h^4\rangle'_{1234}
 ={}&-\frac{g_t^4}{2}\sum_{\aa_1,\ldots,\aa_4=\pm1}
 \left(\prod_{v=1}^4\aa_v\right)
 \int_{-\infty}^0\prod_{v=1}^4
 [\di\tau_v\,a(\tau_v)G_{\aa_v}(k_v;\tau_v)]\nonumber\\
 &\times\int\frac{\di^3\mb q}{(2\pi)^3}
 \sum_{r_1,\ldots,r_4}
 \tr_2[D^{12}J_{r_2}D^{23}J_{r_3}D^{34}J_{r_4}D^{41}J_{r_1}],
 \label{eq:top-box-integral}
\end{align}
where $D^{ij}=D^{r_ir_j}_{\aa_i\aa_j}(\mb p_{ij};\tau_i,\tau_j)$
and $\tr_2$ contracts the two spinor components. The sum over
$r_1,\ldots,r_4$ includes all assignments of the two Yukawa vertex types
for this fixed ordering. 

Choose the successive momenta as $\mb q+\mb k_1$,
$\mb K=\mb q+\mb k_s$, $\mb q-\mb k_4$, and $\mb q$.
In the double-soft region $q,K\sim k_s$, it is useful to introduce
centered hard momenta,
\begin{equation}
 \mb k_L=\frac{\mb k_1-\mb k_2}{2},\qquad
 \mb k_R=\frac{\mb k_3-\mb k_4}{2},\qquad
 \bm\ell=\mb q+\frac{\mb k_s}{2}.
 \label{eq:top-centered-momenta}
\end{equation}
The hard lines then carry $\mb k_L+\bm\ell$ and
$\mb k_R+\bm\ell$. In particular the right momentum is exactly
$\mb q-\mb k_4=\mb q+\mb k_s+\mb k_3$.
At leading collapsed order, $k_L\simeq k_{12}/2$ and
$k_R\simeq k_{34}/2$.

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{figure/quark box NL.pdf}
    \caption{The leading nonlocal part of the top-quark box with cyclic
    ordering $1234$. Cutting the two soft lines separates two tree
    subgraphs, each containing one hard fermion propagator. The central
    factor contains the nonlocal parts of the two soft propagators and
    their loop momentum integral. Spinor indices are contracted between
    the three factors.}
    \label{fig:top-cut}
\end{figure}

The corresponding factorization is shown in Fig.~\ref{fig:top-cut}.
We take the branch $\eta=+1$ for example, and attaching the soft
endpoints to each hard line gives matrices $V_{12}$ and $V_{34}$.
Their normalization is fixed in App.~\ref{app:top-soft} so that
the cut spin factor of this ordering is
\begin{equation}
 \frac{4^{-2\ii\tilde\nu}}{(4\pi)^2}
 \Gamma^4(\tfrac12-\ii\tilde\nu)
 (qK)^{2\ii\tilde\nu}\prod_{v=1}^4(-\tau_v)^{\ii\tilde\nu}
 \tr_2[V_{12}\sigma_KV_{34}\sigma_q].
 \label{eq:top-cut-trace}
\end{equation}
The four orderings contributing to the collapsed $s$ channel are
$1234,1243,1342,1432$. They replace the trace by
\begin{equation}
 \tr_2[(V_{12}+V_{21})\sigma_K(V_{34}+V_{43})\sigma_q].
 \label{eq:top-four-orderings}
\end{equation}
Every time and SK label remains attached to its physical external
vertex, so all four terms have the same exact Higgs factors.

At zeroth soft order, the vertex-type sum at a hard end is
proportional to its direction. Writing its scalar coefficient as
$F_{0;\aa\bb}$, one finds
\begin{align}
 &V_{12}^{(0)}=-F_{0;\aa_1\aa_2}(k_L;\tau_1,\tau_2)
                  \bm\sigma\cdot\mb n_L,
 &&V_{21}^{(0)}=+F_{0;\aa_1\aa_2}(k_L;\tau_1,\tau_2)
                  \bm\sigma\cdot\mb n_L,\nonumber\\
 &V_{12}^{(0)}+V_{21}^{(0)}=0,
 &&V_{34}^{(0)}+V_{43}^{(0)}=0,
 \label{eq:top-cancellation}
\end{align}
where $\mb n_L=\widehat{\mb k}_L$ and
$\mb n_R=\widehat{\mb k}_R$. The scalar coefficient is invariant
under reversal of its two time and SK labels. 
Thus the cancellation holds at the integrand level. 
A single ordering 
has a nonzero zeroth-order trace, but the sum removes the candidate
$k_s^{3\pm4\ii\tilde\nu}$ oscillatory signal.

The first-order corrections to both the hard and soft propagators
contribute to the hard-end matrices.
Their sum can be organized as
\begin{align}
 V_{12}^{(1)}+V_{21}^{(1)}&=-\frac2{k_L}\bm\sigma\cdot\mb v_L,
 &\mb v_L&=\Pi^L_1\bm\ell+
                  \Pi^L_2\mb n_L(\mb n_L\cdot\bm\ell),\nonumber\\
 V_{34}^{(1)}+V_{43}^{(1)}&=-\frac2{k_R}\bm\sigma\cdot\mb v_R,
 &\mb v_R&=\Pi^R_1\bm\ell+
                  \Pi^R_2\mb n_R(\mb n_R\cdot\bm\ell).
 \label{eq:top-first-order}
\end{align}
The two scalar functions $\Pi_i$ depend on the hard momentum, the endpoint
times and SK labels, and their explicit expressions are given in
\eqref{eq:top-Pi}.
The function $\Pi_1$ includes the hard-direction correction and
the two soft-endpoint corrections, and $\Pi_2$ contains the
hard-magnitude derivative and its associated direction correction.
Substituting \eqref{eq:top-first-order} into \eqref{eq:top-four-orderings}
and using $\tr_2[(\bm\sigma\cdot\mb a)(\bm\sigma\cdot\mb b)
(\bm\sigma\cdot\mb c)(\bm\sigma\cdot\mb d)]
=2[(\mb a\cdot\mb b)(\mb c\cdot\mb d)
-(\mb a\cdot\mb c)(\mb b\cdot\mb d)
+(\mb a\cdot\mb d)(\mb b\cdot\mb c)]$, the leading term of the sum of the four
traces becomes:
\begin{align}
 &\tr_2[(V_{12}^{(1)}+V_{21}^{(1)})\sigma_K
                    (V_{34}^{(1)}+V_{43}^{(1)})\sigma_q]
 \nonumber\\
 &\quad=\frac{8}{k_Lk_RqK}\left[
 2(\mb v_L\cdot\bm\ell)(\mb v_R\cdot\bm\ell)
 -\frac12(\mb v_L\cdot\mb k_s)(\mb v_R\cdot\mb k_s)
 -(\mb v_L\cdot\mb v_R)(\ell^2-k_s^2/4)\right].
 \label{eq:top-scalar-trace}
\end{align}
The factor $8$ consists of $2$ from the Pauli identity and $4$
from the two ordering sums. 
Consequently the first nonzero
oscillatory signal scales as $k_s^{5\pm4\ii\tilde\nu}$.

\subsection{Factorized Loop and Time Integrals}
\label{sec:top-factorized}
The scalar trace in \eqref{eq:top-scalar-trace} reduces the
remaining loop to a scalar integral. We define its dimensionless
coefficients separately for each common soft branch $\eta=\pm1$ by
\begin{align}
 &\int\frac{\di^3\mb q}{(2\pi)^3}(qK)^{2\ii\eta\tilde\nu-1}
 \left[2(\mb v_{L,\eta}\cdot\bm\ell)(\mb v_{R,\eta}\cdot\bm\ell)
 -\frac12(\mb v_{L,\eta}\cdot\mb k_s)(\mb v_{R,\eta}\cdot\mb k_s)
 -(\mb v_{L,\eta}\cdot\mb v_{R,\eta})(\ell^2-k_s^2/4)\right]\nonumber\\
 &\hspace{20mm}=\frac{k_s^{5+4\ii\eta\tilde\nu}}{(4\pi)^{3/2}}
 \sum_{i,j=1}^2\mathcal I^{ij}_{t}
 (-\ii\eta\tilde\nu,\theta_1,\theta_3,\phi_3)\Pi^L_{i,\eta}\Pi^R_{j,\eta}.
 \label{eq:top-loop-definition}
\end{align}
Here $\mb v_{A,\eta}=\Pi^A_{1,\eta}\bm\ell
+\Pi^A_{2,\eta}\mb n_A(\mb n_A\cdot\bm\ell)$ for $A=L,R$,
with $\Pi_{i,\eta}$ defined in \eqref{eq:top-Pi}.
The choice $\eta=+1$ reproduces the preceding derivation.
For $\eta=-1$, the imaginary parts of the soft exponents and the loop argument change sign,
while the hard propagators entering $\Pi_{i,\eta}$ retain their
mass parameter and SK labels. 
% Every polynomial momentum moment in \eqref{eq:top-loop-definition}
% reduces to the scalar convolution $f_0$ of \eqref{eq:f0}.
The resulting coefficients satisfy
\begin{equation}
 \mathcal I^{11}_{t}
 =3\mathcal I^{12}_{t}
 =3\mathcal I^{21}_{t}.
 \label{eq:top-loop-relations}
\end{equation}
These three coefficients are independent of angles. All angular
dependence is carried by $\mathcal I^{22}_{t}$, whose
explicit expression is collected in App.~\ref{app:top-soft}.

For the same-branch oscillatory contribution, the nonlocal part
of the loop integral in \eqref{eq:top-box-integral}, summed over
the four orderings, is
\begin{align}
 &\left[\int\frac{\di^3\mb q}{(2\pi)^3}
 \sum_{\mathrm{ord}}\sum_{r_1,\ldots,r_4}
 \tr_2[D^{12}J_{r_2}D^{23}J_{r_3}D^{34}J_{r_4}D^{41}J_{r_1}]
 \right]_{\mathrm{NonL}}
 \nonumber\\
 &\quad\simeq \sum_{\eta=\pm1} 
 \frac{8\,4^{-2\ii\eta\tilde\nu}\Gamma^4(\tfrac12-\ii\eta\tilde\nu)
 k_s^{5+4\ii\eta\tilde\nu}}{(4\pi)^{7/2}k_Lk_R}
 \prod_{v=1}^4(-\tau_v)^{\ii\eta\tilde\nu}
 \nonumber\\
 &\qquad\times
 \sum_{i,j=1}^2\mathcal I^{ij}_{t}
 (-\ii\eta\tilde\nu,\theta_1,\theta_3,\phi_3)
 \Pi_{i,\eta}^L\Pi_{j,\eta}^R.
 \label{eq:top-cut-loop}
\end{align}
Here $\sum_{\mathrm{ord}}$ acts on the loop as in
\eqref{eq:top-four-orderings}.

The two time integrations on each side are now independent.
For the left subgraph, the leading collapsed term at
$k_1\simeq k_2\simeq k_L\simeq k_{12}/2$ defines
\begin{align}
 &\sum_{\aa,\bb=\pm1}\aa\bb
 \int_{-\infty}^0 \frac{\di\tau_1}{H\tau_1} \frac{\di\tau_2}{H\tau_2} \,
 (-\tau_1)^{\ii\eta\tilde\nu}(-\tau_2)^{\ii\eta\tilde\nu}
 G_{\aa}(k_1;\tau_1)G_{\bb}(k_2;\tau_2)
 \Pi_{i,\eta;\aa\bb}(k_{12}/2;\tau_1,\tau_2)
 \nonumber\\
 &\hspace{20mm}\simeq\frac{H^2}{4k_1^3k_2^3}
 k_{12}^{-2\ii\eta\tilde\nu}\mathcal P_{t,i,\eta}(\tilde\nu),
 \qquad i=1,2,\quad\eta=\pm1.
 \label{eq:top-time-definition}
\end{align}
Below, $\mathcal P_{t,i}\equiv\mathcal P_{t,i,+}$
denotes the positive-branch coefficient.
The right subgraph follows by $(1,2,L)\to(3,4,R)$ and $k_{12}\to k_{34}$.
These time integrals reduce to the Whittaker seed of
Ref.~\cite{qin_closed-form_2023}, denoted here by
$\mathcal K_{\aa\bb}^{\kappa;p_1p_2}$ with $\kappa=\pm1/2$.
% Its scalar kernel has a different normalization from
% $D^{(\mathrm{s})}$ in \eqref{eq:scalar-propagator}.
The seed normalization, its closed form and the combinations giving
$\mathcal P_{t,i}$ are given in App.~\ref{app:top-seed}.

\subsection{Oscillatory Signal}
\label{sec:top-clock}
We now restore the sector and color multiplicities in the final signals.
The two identical spinor sectors supply a factor of $2$, and the color
sum supplies $3$, giving an overall factor $6$ relative to the
single-sector, fixed-color normalization in \eqref{eq:top-box-integral}.
Combining the soft loop and the two hard subgraphs, and adding the
complex-conjugate branch, gives
\begin{align}
 \left[\langle\delta h^4\rangle'_{t,s}\right]_{\mathrm{osc}}
 \simeq{}&-\frac{3g_t^4H^4k_s^5}
 {4k_1^3k_2^3k_3^3k_4^3k_{12}k_{34}}\left[
 \left(\frac{k_s^2}{4k_{12}k_{34}}\right)^{2\ii\tilde\nu}
 C_{t}(\tilde\nu,\theta_1,\theta_3,\phi_3)
 +\text{c.c.}\right],
 \label{eq:top-clock-signal}
\end{align}
where
\begin{equation}
 C_{t}
 =\frac{8\Gamma^4(\tfrac12-\ii\tilde\nu)}{(4\pi)^{7/2}}
 \sum_{i,j=1}^2\mathcal P_{t,i}\mathcal P_{t,j}
 \mathcal I^{ij}_{t}
 (-\ii\tilde\nu,\theta_1,\theta_3,\phi_3).
 \label{eq:top-clock-coefficient}
\end{equation}
% All four $s$-channel orderings are included. With the time coefficients
% normalized to $k_{12}$ and $k_{34}$ in \eqref{eq:top-time-definition},
% the soft phase $4^{-2\ii\tilde\nu}$ is included entirely in the
% momentum ratio in \eqref{eq:top-clock-signal}.

The angular decomposition takes the same form as
\eqref{eq:box-angular-expansion}:
\begin{align}
 C_{t}={}&A_{1,t}
 +A_{2,t}(\cos^2\theta_1+\cos^2\theta_3)
 +A_{3,t}\cos^2\theta_1\cos^2\theta_3\nonumber\\
 &+B_{t}\sin(2\theta_1)\sin(2\theta_3)\cos\phi_3
 +D_{t}\sin^2\theta_1\sin^2\theta_3\cos(2\phi_3).
 \label{eq:top-angular}
\end{align}
The five coefficients depend only on $\tilde\nu$ and satisfy
$A_{3,t}=-3A_{2,t}$. Their analytic expressions are given in
App.~\ref{app:top-coefficients}. In particular all nonconstant
angular terms are proportional to $\mathcal P_{t,2}^2$.
Figure~\ref{fig:top-angular-coefficients} shows the mass dependence
of the five angular coefficients.
% The result is even under either Higgs-pair exchange and under parity.

At fixed angles the oscillatory bracket in
\eqref{eq:top-clock-signal} is
\begin{equation}
 2|C_{t}|\cos\left[
 2\tilde\nu\log\!\left(\frac{k_s^2}{4k_{12}k_{34}}\right)
 +\arg C_{t}\right].
 \label{eq:top-clock-phase}
\end{equation}
The frequency in $\log k_s$ is $4m_t/H$.
Compared with the gauge-boson signals, the top oscillation carries
an additional soft suppression $k_s^2/(k_{12} k_{34})$ from
\eqref{eq:top-cancellation}.

    \caption{The five top quark angular coefficients in
    \eqref{eq:top-angular} as functions of $\tilde\nu=m_t/H$.
    The curves show the natural logarithms of their absolute values.}
    \label{fig:top-angular-coefficients}
\end{figure}

\subsection{Bubble Coefficients}
\label{app:bubble-polarizations}
For $x=-\ii\eta\tilde\nu$, the relative weight of a longitudinal
nonlocal propagator is
\begin{equation}
    \beta(x)\equiv\frac{(x+1/2)^2}{1/4-x^2}
    =\frac{1+2x}{1-2x}.
    \label{eq:bubble-longitudinal-weight}
\end{equation}
Define $\mathcal I_{\sigma_1\sigma_2}(x)$ by restricting the loop in
\eqref{eq:bubble-cut-loop} to the soft line polarizations
$(\sigma_1,\sigma_2)$ and replacing $\mathcal I_{\mathrm{Bub}}$ on its
right-hand side by $\mathcal I_{\sigma_1\sigma_2}$. The independent
coefficients are
\begin{align}
    \mathcal I_{LL}(x)
    &=[\beta(x)]^2
    \left[\mathcal J_0(2,2;x)+2\mathcal J_1(1,2;x)
    +\mathcal J_2(0,2;x)\right],
    \label{eq:bubble-ILL}\\
    \mathcal I_{L+}(x)
    &=\frac{\beta(x)}2
    \left[\mathcal J_0(0,2;x)-\mathcal J_2(0,2;x)\right],
    \label{eq:bubble-ILplus}\\
    \mathcal I_{++}(x)
    &=\frac14\left[
    \mathcal J_0(0,0;x)+\mathcal J_0(2,2;x)
    +\mathcal J_2(0,2;x)\right.\nonumber\\
    &\hspace{12mm}\left.
    +2\mathcal J_0(1,1;x)+2\mathcal J_1(1,2;x)
    +2\mathcal J_1(0,1;x)\right],
    \label{eq:bubble-Iplusplus}\\
    \mathcal I_{+-}(x)
    &=\frac14\left[
    \mathcal J_0(0,0;x)+\mathcal J_0(2,2;x)
    +\mathcal J_2(0,2;x)\right.\nonumber\\
    &\hspace{12mm}\left.
    -2\mathcal J_0(1,1;x)+2\mathcal J_1(1,2;x)
    -2\mathcal J_1(0,1;x)\right].
    \label{eq:bubble-Iplusminus}
\end{align}
The remaining entries satisfy
$\mathcal I_{--}=\mathcal I_{++}$,
$\mathcal I_{-+}=\mathcal I_{+-}$, and
$\mathcal I_{L-}=\mathcal I_{+L}=\mathcal I_{-L}=\mathcal I_{L+}$.
Consequently,
\begin{equation}
    \mathcal I_{\mathrm{Bub}}
    =\mathcal I_{LL}+4\mathcal I_{L+}
    +2\mathcal I_{++}+2\mathcal I_{+-},
    \label{eq:bubble-channel-sum}
\end{equation}
which reduces to \eqref{eq:bubble-polarization-sum} using the moment
relations in App.~\ref{app:loop-integrals}.

For the $\eta=+1$ branch, the fully reduced sum is
\begin{equation}
    \mathcal I_{\mathrm{Bub}}(-\ii\tilde\nu)
    =\frac{3\cdot2^{-4-4\ii\tilde\nu}(3\ii-2\tilde\nu)^2
    \Gamma(-3/2-2\ii\tilde\nu)\Gamma(3+2\ii\tilde\nu)
    \sinh^2(\pi\tilde\nu)}
    {\pi[1+(3\ii-2\tilde\nu)\tilde\nu]^2}.
    \label{eq:bubble-I-closed}
\end{equation}
Substituting this into \eqref{eq:bubble-coefficient} gives
\begin{align}
    C^{\mathrm{Bub}}(\tilde\nu)
    ={}&\frac{3(\tilde\nu-\ii)(\tilde\nu-2\ii)^4}
    {256\pi^{7/2}\tilde\nu^2(2\tilde\nu-\ii)}
    \Gamma^2(-2-\ii\tilde\nu)\Gamma^2(5/2+\ii\tilde\nu)
    \nonumber\\
    &\times\Gamma(-3/2-2\ii\tilde\nu)
    \Gamma(1+2\ii\tilde\nu)\sinh^2(\pi\tilde\nu).
    \label{eq:bubble-C-closed}
\end{align}
For real $\tilde\nu>0$,
the opposite-branch expression is its complex conjugate.

\subsection{Triangle Coefficients}
\label{app:triangle-coefficients}
\subsubsection{Polarization sums}
Let $x=-\ii\eta\tilde\nu$ and let $\beta(x)$ be the longitudinal
weight in \eqref{eq:bubble-longitudinal-weight}. Summing the nine
soft-polarization combinations gives
\begin{align}
    \mathcal I^L_{\mathrm{Tri}}(x,\theta)
    ={}&[\beta(x)]^2\left\{
    \frac{\sin^2\theta}{2}\left[
    \mathcal J_0(2,2;x)+\mathcal J_1(1,2;x)
    -\mathcal J_2(2,2;x)-\mathcal J_3(1,2;x)\right]\right.
    \nonumber\\
    &\hspace{15mm}\left.+\cos^2\theta\left[
    \mathcal J_2(2,2;x)+\mathcal J_3(1,2;x)
    +\mathcal J_1(1,2;x)+\mathcal J_2(0,2;x)\right]\right\}
    \nonumber\\
    &+\frac{\beta(x)(1+\cos^2\theta)}2
    \left[\mathcal J_0(0,2;x)-\mathcal J_2(0,2;x)\right]
    \nonumber\\
    &+\frac{\sin^2\theta}{2}\left[
    \mathcal J_2(2,2;x)+\mathcal J_3(1,2;x)
    +\mathcal J_1(1,2;x)+\mathcal J_2(0,2;x)
    +\mathcal J_0(0,0;x)\right]
    \nonumber\\
    &+\cos^2\theta\left[
    \mathcal J_0(2,2;x)-\mathcal J_2(2,2;x)
    +\mathcal J_1(1,2;x)-\mathcal J_3(1,2;x)\right].
    \label{eq:triangle-IL-explicit}
\end{align}
The terms proportional to $\beta^2$, $\beta$, and $1$ arise from
two longitudinal, mixed, and two transverse soft lines, respectively.

For a hard line with helicity $+$, the corresponding expression is
\begin{align}
    \mathcal I^+_{\mathrm{Tri}}(x,\theta)
    ={}&[\beta(x)]^2\left\{
    \frac{1+\cos^2\theta}{4}\left[
    \mathcal J_0(2,2;x)-\mathcal J_2(2,2;x)
    +\mathcal J_1(1,2;x)-\mathcal J_3(1,2;x)\right]\right.
    \nonumber\\
    &\hspace{15mm}\left.+\frac{\sin^2\theta}{2}\left[
    \mathcal J_2(2,2;x)+\mathcal J_1(1,2;x)
    +\mathcal J_3(1,2;x)+\mathcal J_2(0,2;x)\right]\right\}
    \nonumber\\
    &+\frac{\beta(x)(2+\sin^2\theta)}4
    \left[\mathcal J_0(0,2;x)-\mathcal J_2(0,2;x)\right]
    \nonumber\\
    &+\frac{1+\cos^2\theta}{4}\left[
    \mathcal J_2(2,2;x)+\mathcal J_1(1,2;x)
    +\mathcal J_3(1,2;x)+\mathcal J_2(0,2;x)
    +\mathcal J_0(0,0;x)\right]
    \nonumber\\
    &+\frac{\sin^2\theta}{2}\left[
    \mathcal J_0(2,2;x)-\mathcal J_2(2,2;x)
    +\mathcal J_1(1,2;x)-\mathcal J_3(1,2;x)\right].
    \label{eq:triangle-IP-explicit}
\end{align}
At zero chemical potential, reducing the Gamma functions gives the same
result for all three hard-line polarizations:
\begin{align}
    \mathcal I^L_{\mathrm{Tri}}(-\ii\tilde\nu,\theta)
    &=\mathcal I^+_{\mathrm{Tri}}(-\ii\tilde\nu,\theta)
    =\mathcal I^-_{\mathrm{Tri}}(-\ii\tilde\nu,\theta)
    =\frac13\mathcal I_{\mathrm{Bub}}(-\ii\tilde\nu)
    \nonumber\\
    &=\frac{2^{-4-4\ii\tilde\nu}(3\ii-2\tilde\nu)^2
    \Gamma(-3/2-2\ii\tilde\nu)\Gamma(3+2\ii\tilde\nu)
    \sinh^2(\pi\tilde\nu)}
    {\pi[1+(3\ii-2\tilde\nu)\tilde\nu]^2}.
    \label{eq:triangle-I-closed}
\end{align}
Summing over the hard-line polarizations reproduces
$\mathcal I_{\mathrm{Bub}}$ by polarization completeness.
Thus the angular dependence of the separate soft-polarization
contributions cancels in their sum. The coefficient of a single
triangle reduces to
\begin{equation}
    C^{\mathrm{Tri}}(\tilde\nu)
    =\frac{\Gamma^4(-\ii\tilde\nu)}{3(4\pi)^{7/2}}
    \mathcal P^{(1)}_{+}(\tilde\nu)
    \left[\mathcal P^{(2),L}_{+}(\tilde\nu)
    +2\mathcal P^{(2),+}_{+}(\tilde\nu)\right]
    \mathcal I_{\mathrm{Bub}}(-\ii\tilde\nu).
    \label{eq:triangle-C-reduced}
\end{equation}

\subsubsection{Time integral coefficients}
For the scalar propagator of \eqref{eq:scalar-propagator},
the double-folded seed is defined by
\begin{equation}
    \mathcal M_{\aa_1\aa_2}^{p_1p_2}
    \equiv(-\aa_1\aa_2)k^{5+p_{12}}
    \int_{-\infty}^0\di\tau_1\di\tau_2\,
    (-\tau_1)^{p_1}(-\tau_2)^{p_2}
    e^{\ii\aa_1 k\tau_1+\ii\aa_2 k\tau_2}
    \frac{D^{(\mathrm{s})}_{\aa_1\aa_2}(k;\tau_1,\tau_2)}{H^2}.
    \label{eq:triangle-M-definition}
\end{equation}
This is the seed $\widetilde{\mathcal I}_{\aa_1\aa_2}^{p_1p_2}(1,1)$
of Ref.~\cite{qin_closed-form_2023}, with $H$ restored explicitly.
All seeds below have the same mass index $\tilde\nu$.
The closed expressions are
\begin{align}
    \mathcal M_{\pm\mp}^{p_1p_2}
    &=\frac{e^{\mp\ii\pi\bar p_{12}/2}}{2^{5+p_{12}}}
    \prod_{j=1}^{2}
    \frac{\Gamma(5/2+p_j-\ii\tilde\nu)
    \Gamma(5/2+p_j+\ii\tilde\nu)}{\Gamma(3+p_j)},
    \label{eq:triangle-M-opposite}\\
    \mathcal M_{\pm\pm}^{p_1p_2}
    &=\frac{\pm\ii e^{\mp\ii\pi p_{12}/2}e^{-\pi\tilde\nu}}
    {2^{5+p_{12}}}
    \prod_{j=1}^{2}
    \frac{\Gamma(5/2+p_j-\ii\tilde\nu)
    \Gamma(5/2+p_j+\ii\tilde\nu)}{\Gamma(3+p_j)}
    \nonumber\\
    &\quad-\frac{e^{\mp\ii\pi p_{12}/2}}{2^{5+p_{12}}}
    \Gamma(5+p_{12})\Gamma(5/2+p_1\pm\ii\tilde\nu)
    \Gamma(5/2+p_2\pm\ii\tilde\nu)
    \nonumber\\
    &\qquad\times{}_3\widetilde F_2\left[
    \begin{matrix}
    5+p_{12},\;1/2\pm\ii\tilde\nu,\;1\\
    7/2+p_1\pm\ii\tilde\nu,\;7/2+p_2\pm\ii\tilde\nu
    \end{matrix}\middle|1\right].
    \label{eq:triangle-M-same}
\end{align}
% Here ${}_3\widetilde F_2(a,b,c;d,e;1)
% ={}_3F_2(a,b,c;d,e;1)/[\Gamma(d)\Gamma(e)]$ is the regularized
% hypergeometric function.

Set $p_\epsilon=-5/2+\epsilon+\ii\eta\tilde\nu$.
The two transverse coefficients are
\begin{align}
    \mathcal P^{(2),\pm}_{\eta}(\tilde\nu)
    ={}&2^{2\ii\eta\tilde\nu}
    \underset{\epsilon=0}{\operatorname{FP}}
    \sum_{\aa_1,\aa_2=\pm1}
    \left[
    \mathcal M_{\aa_1\aa_2}^{p_\epsilon,p_\epsilon}
    +\ii\aa_1\mathcal M_{\aa_1\aa_2}^{p_\epsilon+1,p_\epsilon}
    +\ii\aa_2\mathcal M_{\aa_1\aa_2}^{p_\epsilon,p_\epsilon+1}
    -\aa_1\aa_2\mathcal M_{\aa_1\aa_2}^{p_\epsilon+1,p_\epsilon+1}
    \right].
    \label{eq:triangle-P2-transverse}
\end{align}
The longitudinal coefficient can be written using the same transverse
combination:
\begin{align}
    \mathcal P^{(2),L}_{\eta}(\tilde\nu)
    ={}&\frac{\mathcal P^{(1)}_{\eta}(\tilde\nu)
    +(1/2+\ii\eta\tilde\nu)^2
    \mathcal P^{(2),+}_{\eta}(\tilde\nu)}{\tilde\nu^2+1/4}
    \nonumber\\
    &+\frac{2^{2\ii\eta\tilde\nu}}{\tilde\nu^2+1/4}
    \underset{\epsilon=0}{\operatorname{FP}}
    \sum_{\aa_1,\aa_2=\pm1}
    \left\{(1/2+\ii\eta\tilde\nu)
    \left[
    \mathcal M_{\aa_1\aa_2}^{p_\epsilon+2,p_\epsilon}
    +\ii\aa_2\mathcal M_{\aa_1\aa_2}^{p_\epsilon+2,p_\epsilon+1}
    \right.\right.\nonumber\\
    &\hspace{30mm}\left.\left.
    +\mathcal M_{\aa_1\aa_2}^{p_\epsilon,p_\epsilon+2}
    +\ii\aa_1\mathcal M_{\aa_1\aa_2}^{p_\epsilon+1,p_\epsilon+2}
    \right]
    +\mathcal M_{\aa_1\aa_2}^{p_\epsilon+2,p_\epsilon+2}
    \right\}.
    \label{eq:triangle-P2-longitudinal}
\end{align}
The operator $\operatorname{FP}_{\epsilon=0}$ selects the constant
coefficient of the Laurent expansion after assembling the indicated
seed sum. The first term proportional to
$\mathcal P^{(1)}_{\eta}$ in \eqref{eq:triangle-P2-longitudinal} is
the contact contribution from differentiating the time-ordering
step functions.
For the positive branch, the finite coefficients can be reduced further.
Introduce four functions at unit argument,
\begin{align}
 \mathcal F_{g,1}&={}_3F_2\!\left[
 \begin{matrix}1,\ \tfrac12+\ii\tilde\nu,\ 2\ii\tilde\nu\\
 1+2\ii\tilde\nu,\ 1+2\ii\tilde\nu\end{matrix}\middle|1\right],\nonumber\\
 \mathcal F_{g,2}&=\left.\partial_d\,
 {}_3\widetilde F_2\!\left[
 \begin{matrix}1+2\ii\tilde\nu,\ \tfrac12-\ii\tilde\nu,\ 1\\
 d,\ 2\end{matrix}\middle|1\right]\right|_{d=1},\nonumber\\
 \mathcal F_{g,3}&=\left.\partial_d\,
 {}_3\widetilde F_2\!\left[
 \begin{matrix}2+2\ii\tilde\nu,\ \tfrac12-\ii\tilde\nu,\ 1\\
 d,\ 3\end{matrix}\middle|1\right]\right|_{d=1},\nonumber\\
 \mathcal F_{g,4}&=\left.\partial_d\partial_e\,
 {}_3\widetilde F_2\!\left[
 \begin{matrix}2\ii\tilde\nu,\ \tfrac12-\ii\tilde\nu,\ 1\\
 d,\ e\end{matrix}\middle|1\right]\right|_{d=e=1}.
 \label{eq:gauge-finite-masters}
\end{align}
The derivatives act on the lower parameters, including these Gamma factors.
For compactness, write
\begin{equation}
 w=e^{\pi\tilde\nu},\qquad
 R=\frac{\Gamma(2\ii\tilde\nu)}{\Gamma^2(\tfrac12+\ii\tilde\nu)},\qquad
 \Psi=\psi(2\ii\tilde\nu),\qquad
 \Xi=\psi^{(1)}(\tfrac12+\ii\tilde\nu),
 \label{eq:gauge-finite-abbreviations}
\end{equation}
where $\psi$ and $\psi^{(1)}$ are the digamma and trigamma functions.
The finite transverse combination is
\begin{align}
 \mathcal T_g={}&R\Bigg\{
 (1-w^{-2})\left[\Psi^2-\Xi
 -\frac{4\ii\tilde\nu\Psi+2}{1+2\ii\tilde\nu}\right]
 -\frac{4\ii\pi}{1+w^2}
 \left[\Psi-\frac{2\ii\tilde\nu}{1+2\ii\tilde\nu}\right]
 +\pi^2\left[1+\frac{4(w^2-1)}{(1+w^2)^2}\right]\Bigg\}
 \nonumber\\
 &-\frac{\ii[(2+\ii\tilde\nu)w-\ii\tilde\nu/w]}{1+2\ii\tilde\nu}
 +\frac{\ii w}{4\tilde\nu^2}\mathcal F_{g,1}
 -\frac{2\tilde\nu}{w}\mathcal F_{g,2}
 +\frac{\ii}{w}\mathcal F_{g,4}.
 \label{eq:gauge-finite-T}
\end{align}
The remaining longitudinal contribution is
\begin{align}
 \mathcal D_g={}&\frac{4R}{(1-2\ii\tilde\nu)(3+2\ii\tilde\nu)}
 \left\{(1-w^{-2})\left[
 \frac{1+4\ii\tilde\nu}{1+2\ii\tilde\nu}+2\ii\tilde\nu\Psi\right]
 +\frac{4\pi\tilde\nu}{1+w^2}\right\}
 \nonumber\\
 &+\frac{2\ii}{(1-2\ii\tilde\nu)(3+2\ii\tilde\nu)}
 \left[(8+10\ii\tilde\nu-3\tilde\nu^2)w
 -\frac{6+8\ii\tilde\nu-3\tilde\nu^2}{w}\right]
 \nonumber\\
 &+\frac{2\tilde\nu(1+2\ii\tilde\nu)}{(1-2\ii\tilde\nu)w}
 \mathcal F_{g,3}.
 \label{eq:gauge-finite-D}
\end{align}
The coefficients after taking the finite part are therefore
\begin{align}
 \mathcal P^{(2),\pm}_{+}(\tilde\nu)
 &=\Gamma(2\ii\tilde\nu)\mathcal T_g,\nonumber\\
 \mathcal P^{(2),L}_{+}(\tilde\nu)
 &=\Gamma(2\ii\tilde\nu)
 \left[\frac{1+2\ii\tilde\nu}{1-2\ii\tilde\nu}\mathcal T_g
 +\mathcal D_g\right].
 \label{eq:gauge-P2-finite}
\end{align}
% These expressions include the longitudinal contact term and retain the
% parameter derivatives in \eqref{eq:gauge-finite-masters} exactly.
The negative branch is obtained by complex conjugation.

\subsection{Box Coefficients}
\label{app:box-coefficients}
\subsubsection{Polarization sums}
Let $\lambda_1,\lambda_3=L,\pm$ denote the polarizations of the
hard lines with momenta $\mb k_1$ and $\mb k_3$. With
$\mb K=\mb q+\mb k_s$, define the soft-polarization sum by
\begin{align}
    &\sum_{\sigma_1,\sigma_2=L,\pm}
    [\beta(x)]^{\delta_{\sigma_1L}+\delta_{\sigma_2L}}
    \int\frac{\di^3\mb q}{(2\pi)^3}
    \left(\frac q2\right)^{-2x}
    \left(\frac K2\right)^{-2x}
    (e_{\mb k_1}^{(\lambda_1)*}\!\cdot e_{\mb K}^{(\sigma_1)})
    (e_{\mb K}^{(\sigma_1)*}\!\cdot e_{\mb k_3}^{(\lambda_3)})
    \nonumber\\
    &\qquad\times
    (e_{\mb k_3}^{(\lambda_3)*}\!\cdot e_{\mb q}^{(\sigma_2)})
    (e_{\mb q}^{(\sigma_2)*}\!\cdot e_{\mb k_1}^{(\lambda_1)})
    =\frac{4^{2x}k_s^{3-4x}}{(4\pi)^{3/2}}
    \mathcal I^{\lambda_1\lambda_3}_{\mathrm{Box}}
    (x,\theta_1,\theta_3,\phi_3).
    \label{eq:box-I-definition}
\end{align}
The angles are defined by the convention stated below
\eqref{eq:collapsed-kinematics}. The two transverse hard-line coefficients
$\mathcal P^{(2),+}_{\eta}=\mathcal P^{(2),-}_{\eta}$ are equal, so it
suffices to retain three groups:
\begin{align}
    \mathcal I^{LH}_{\mathrm{Box}}
    &\equiv\sum_{\lambda=\pm}
    \left(\mathcal I^{L\lambda}_{\mathrm{Box}}
    +\mathcal I^{\lambda L}_{\mathrm{Box}}\right),
    &\mathcal I^{HH}_{\mathrm{Box}}
    &\equiv\sum_{\lambda_1,\lambda_3=\pm}
    \mathcal I^{\lambda_1\lambda_3}_{\mathrm{Box}},
    \label{eq:box-channel-groups}
\end{align}
together with $\mathcal I^{LL}_{\mathrm{Box}}$. Here $H$ in a polarization
superscript denotes the sum over transverse helicities.

For $x=-\ii\tilde\nu$, the longitudinal result is
\begin{align}
    &\mathcal I^{LL}_{\mathrm{Box}}
    (-\ii\tilde\nu,\theta_1,\theta_3,\phi_3)
    \nonumber\\
    &\quad=-\frac{4^{-3-2\ii\tilde\nu}(2\tilde\nu-3\ii)
    \Gamma(-3/2-2\ii\tilde\nu)\Gamma(1+2\ii\tilde\nu)
    \sinh^2(\pi\tilde\nu)}
    {\pi(\tilde\nu-\ii)(\tilde\nu-2\ii)(2\tilde\nu-\ii)}
    \nonumber\\
    &\qquad\times\Big\{
    -16-19\ii\tilde\nu+6\tilde\nu^2
    +\tilde\nu(2\tilde\nu-\ii)
    \big[\cos(2\theta_1)+\cos(2\theta_3)
    +3\cos(2\theta_1)\cos(2\theta_3)\big]
    \nonumber\\
    &\hspace{24mm}
    +4(2\tilde\nu^2-5\ii\tilde\nu-3)
    \cos(2\phi_3)\sin^2\theta_1\sin^2\theta_3
    \nonumber\\
    &\hspace{24mm}
    -8\tilde\nu(\tilde\nu-\ii)\cos\phi_3
    \sin(2\theta_1)\sin(2\theta_3)\Big\}.
    \label{eq:box-ILL-closed}
\end{align}
Reducing the mixed and transverse results to the same Gamma-function
basis gives
\begin{align}
    &\mathcal I^{LH}_{\mathrm{Box}}
    (-\ii\tilde\nu,\theta_1,\theta_3,\phi_3)
    \nonumber\\
    &\quad=-\frac{2\cdot4^{-3-2\ii\tilde\nu}(2\tilde\nu-3\ii)
    \Gamma(-3/2-2\ii\tilde\nu)\Gamma(1+2\ii\tilde\nu)
    \sinh^2(\pi\tilde\nu)}
    {\pi(\tilde\nu-\ii)(\tilde\nu-2\ii)(2\tilde\nu-\ii)}
    \nonumber\\
    &\qquad\times\Big\{
    -32-37\ii\tilde\nu+10\tilde\nu^2
    -\tilde\nu(2\tilde\nu-\ii)
    \big[\cos(2\theta_1)+\cos(2\theta_3)
    +3\cos(2\theta_1)\cos(2\theta_3)\big]
    \nonumber\\
    &\hspace{24mm}
    -4(2\tilde\nu^2-5\ii\tilde\nu-3)
    \cos(2\phi_3)\sin^2\theta_1\sin^2\theta_3
    \nonumber\\
    &\hspace{24mm}
    +8\tilde\nu(\tilde\nu-\ii)\cos\phi_3
    \sin(2\theta_1)\sin(2\theta_3)\Big\},
    \label{eq:box-ILH-explicit}\\[1ex]
    &\mathcal I^{HH}_{\mathrm{Box}}
    (-\ii\tilde\nu,\theta_1,\theta_3,\phi_3)
    \nonumber\\
    &\quad=-\frac{4^{-3-2\ii\tilde\nu}(2\tilde\nu-3\ii)
    \Gamma(-3/2-2\ii\tilde\nu)\Gamma(1+2\ii\tilde\nu)
    \sinh^2(\pi\tilde\nu)}
    {\pi(\tilde\nu-\ii)(\tilde\nu-2\ii)(2\tilde\nu-\ii)}
    \nonumber\\
    &\qquad\times\Big\{
    -64-75\ii\tilde\nu+22\tilde\nu^2
    +\tilde\nu(2\tilde\nu-\ii)
    \big[\cos(2\theta_1)+\cos(2\theta_3)
    +3\cos(2\theta_1)\cos(2\theta_3)\big]
    \nonumber\\
    &\hspace{24mm}
    +4(2\tilde\nu^2-5\ii\tilde\nu-3)
    \cos(2\phi_3)\sin^2\theta_1\sin^2\theta_3
    \nonumber\\
    &\hspace{24mm}
    -8\tilde\nu(\tilde\nu-\ii)\cos\phi_3
    \sin(2\theta_1)\sin(2\theta_3)\Big\}.
    \label{eq:box-IHH-explicit}
\end{align}
These expressions satisfy the relations implied by polarization
completeness and \eqref{eq:triangle-I-closed}:
\begin{align}
    \mathcal I^{LH}_{\mathrm{Box}}
    &=\frac23\mathcal I_{\mathrm{Bub}}-2\mathcal I^{LL}_{\mathrm{Box}},
    \label{eq:box-ILH-closed}\\
    \mathcal I^{HH}_{\mathrm{Box}}
    &=\frac13\mathcal I_{\mathrm{Bub}}+\mathcal I^{LL}_{\mathrm{Box}}.
    \label{eq:box-IHH-closed}
\end{align}
Indeed, all angular terms cancel in
$\mathcal I^{LH}_{\mathrm{Box}}+2\mathcal I^{LL}_{\mathrm{Box}}$ and
$\mathcal I^{HH}_{\mathrm{Box}}-\mathcal I^{LL}_{\mathrm{Box}}$.
The three groups also satisfy
$\mathcal I^{LL}_{\mathrm{Box}}+\mathcal I^{LH}_{\mathrm{Box}}
+\mathcal I^{HH}_{\mathrm{Box}}=\mathcal I_{\mathrm{Bub}}$.

The dimensionless box coefficient is consequently
\begin{align}
    C^{\mathrm{Box}}(\tilde\nu,\theta_1,\theta_3,\phi_3)
    ={}&\frac{\Gamma^4(-\ii\tilde\nu)}{(4\pi)^{7/2}}
    \Big\{
    [\mathcal P^{(2),L}_{+}]^2\mathcal I^{LL}_{\mathrm{Box}}
    +\mathcal P^{(2),L}_{+}\mathcal P^{(2),+}_{+}
    \mathcal I^{LH}_{\mathrm{Box}}
    +[\mathcal P^{(2),+}_{+}]^2\mathcal I^{HH}_{\mathrm{Box}}\Big\}
    \nonumber\\
    ={}&\frac{\Gamma^4(-\ii\tilde\nu)}{(4\pi)^{7/2}}
    \Big\{
    [\mathcal P^{(2),L}_{+}-\mathcal P^{(2),+}_{+}]^2
    \mathcal I^{LL}_{\mathrm{Box}}
    +\frac13\mathcal P^{(2),+}_{+}
    [2\mathcal P^{(2),L}_{+}+\mathcal P^{(2),+}_{+}]
    \mathcal I_{\mathrm{Bub}}\Big\}.
    \label{eq:box-C-reduced}
\end{align}
The opposite nonlocal branch follows by complex conjugation.

For the $\eta=+1$ branch, a soft line factorizes as
\begin{equation}
 D^{r_1 r_2}_{\mathrm{NonL},+}(p;\tau,\tau')
 \simeq-\frac{\Gamma^2(\tfrac12-\ii\tilde\nu)}{4\pi}

 \label{eq:top-endpoint-factorization}
\end{equation}
where, through first order,
\begin{align}

 &L_{\overline{\mathrm L}}&=\epsmat
 \left(\one-\frac{\ii\tau\bm\sigma\cdot\mb p}{1+2\ii\tilde\nu}\right),
 \nonumber\\
 U_{\mathrm L}&=\left(\one+\frac{\ii\tau\bm\sigma\cdot\mb p}{1+2\ii\tilde\nu}\right)\epsmat,
 &U_{\overline{\mathrm L}}&=\one-\frac{\ii\tau\bm\sigma\cdot\mb p}{1+2\ii\tilde\nu}.
 \label{eq:top-endpoints}
\end{align}
The matrices in \eqref{eq:top-cut-trace} are the vertex-type sums
\begin{align}

 V_{21}&=\sum_{r_1,r_2}

 D^{r_2r_1}_{\aa_2\aa_1}(-\mb k_L+\bm\ell;\tau_2,\tau_1)
 J_{r_1}L_{r_1}(K,\tau_1).
 \label{eq:top-hard-blocks}
\end{align}
The right matrices follow by $(L,1,2,q,K)\to(R,3,4,K,q)$.
The full scalar coefficients of the two cut propagators are kept
outside these matrices, as displayed in \eqref{eq:top-cut-trace}.

We define two auxiliary functions $F_0$ and $F_1$ to express the subgraph contributions.
With $X=-p\tau_1$, $Y=-p\tau_2$, their expressions for the soft branches
$\eta=\pm1$ are
\begin{align}

 \label{eq:top-hard-kernels}
\end{align}

\end{equation}

\subsection{Whittaker Seed and Time Integrals}
\label{app:top-seed}
We denote the Whittaker seed $\mathcal I$ of
Ref.~\cite{qin_closed-form_2023} by $\mathcal K$.
For equal external energies $k$, internal magnitude $p=rk$, and
$u=2r/(1+r)$, the integral seed is defined by
\begin{align}
 \mathcal K_{\aa\bb}^{\kappa;p_1,p_2}(u,u)
 \equiv{}&(-\aa\bb)p^{3+p_1+p_2}
 \int_{-\infty}^0\di\tau_1\di\tau_2\,
 (-\tau_1)^{p_1}(-\tau_2)^{p_2}
 e^{\ii\aa k\tau_1+\ii\bb k\tau_2}
 \Dsc{\kappa}_{\aa\bb}(p;\tau_1,\tau_2).
 \label{eq:top-K-definition}
\end{align}
In terms of $x_1=-k\tau_1$, $x_2=-k\tau_2$, its time moment is
\begin{equation}
 \int_0^\infty\di x_1\di x_2\,x_1^{p_1}x_2^{p_2}e^{-\ii\aa x_1-\ii\bb x_2}
 k\Dsc{\kappa}_{\aa\bb}(rk;-x_1/k,-x_2/k)
 =-\aa\bb\,r^{-3-p_1-p_2}\mathcal K_{\aa\bb}^{\kappa;p_1,p_2}(u,u).
 \label{eq:top-K-scaling}
\end{equation}

At $r=1$, write
$\mathcal K_{\aa\bb}^{\kappa;p_1,p_2}
\equiv\mathcal K_{\aa\bb}^{\kappa;p_1,p_2}(1,1)$.
The closed expressions are
\begin{align}
 \mathcal K_{\pm\mp}^{\kappa;p_1,p_2}
 &=\frac{e^{\mp\ii\pi(p_1-p_2)/2+\ii\pi\kappa}}{2^{3+p_1+p_2}}
 \frac{\Gamma[\tfrac32+p_1-\ii\tilde\nu,\tfrac32+p_1+\ii\tilde\nu,
               \tfrac32+p_2-\ii\tilde\nu,\tfrac32+p_2+\ii\tilde\nu]}
 {\Gamma[2+p_1\pm\kappa,2+p_2\mp\kappa]},
 \label{eq:top-K-opposite}\\
 \mathcal K_{\pm\pm}^{\kappa;p_1,p_2}
 &=\frac{\mp\ii e^{\mp\ii\pi(p_1+p_2)/2}}{2^{3+p_1+p_2}}
 \Gamma[\tfrac32+p_1\pm\ii\tilde\nu,\tfrac32+p_2\pm\ii\tilde\nu]
 \nonumber\\
 &\quad\times\left\{
 \frac{e^{2\ii\pi\kappa}+e^{-2\pi\tilde\nu}}{2\pi}
 \frac{\Gamma[\tfrac32+p_1\mp\ii\tilde\nu,\tfrac32+p_2\mp\ii\tilde\nu,
               \tfrac12\pm\kappa-\ii\tilde\nu,\tfrac12\pm\kappa+\ii\tilde\nu]}
 {\Gamma[2+p_1\pm\kappa,2+p_2\pm\kappa]}
 \right.\nonumber\\
 &\qquad\left.
 \pm\ii\Gamma(3+p_1+p_2)
 {}_3\widetilde F_2\!\left[
 \begin{matrix}3+p_1+p_2,\ \tfrac12\mp\kappa\pm\ii\tilde\nu,\ 1\\
 \tfrac52+p_1\pm\ii\tilde\nu,\ \tfrac52+p_2\pm\ii\tilde\nu
 \end{matrix}\middle|1\right]\right\}.
 \label{eq:top-K-same}
\end{align}

Only two scalar time moments are needed to express all these hard
coefficients. Define
\begin{align}
 \mathcal H_j(q_1,q_2;r)
 ={}&\sum_{\aa,\bb=\pm1}\aa\bb
 \int_0^\infty\di x_1\di x_2\,
 x_1^{q_1-1}x_2^{q_2-1}
 (1+\ii\aa x_1)(1+\ii\bb x_2)e^{-\ii\aa x_1-\ii\bb x_2}
 \nonumber\\
 &\times F_{j;\aa\bb}(rk;-x_1/k,-x_2/k),\qquad j=0,1.
 \label{eq:top-H-definition}
\end{align}
Scale invariance makes the right-hand side independent of $k$.
Expanding the external polynomials with indices $m,n=0,1$, put
$p_1=q_1-3/2+m$, $p_2=q_2-3/2+n$.
Integration by parts sends $x_1\partial_{x_1}$ to
$-(p_1+1)+\ii\aa x_1$. Using the definition \eqref{eq:top-K-definition}
therefore gives directly
\begin{align}

 ={}&\ii

 (\ii\aa)^m(\ii\bb)^n r^{-3-p_1-p_2}

 -\frac{\ii(\aa-r)}r\mathcal K_{\aa\bb}^{+;p_1+1,p_2}

 +(2+\ii\tilde\nu)\mathcal P_{t,2}^2\right],\nonumber\\
 A_{2,t}&=2\mathcal B_{t}\mathcal P_{t,2}^2
 (1+\ii\tilde\nu)^2(1+2\ii\tilde\nu),
 \qquad A_{3,t}=-3A_{2,t},\nonumber\\
 B_{t}&=\mathcal B_{t}\mathcal P_{t,2}^2
       (1+\ii\tilde\nu)(1+2\ii\tilde\nu)(3+2\ii\tilde\nu),\nonumber\\
 D_{t}&=-2\mathcal B_{t}\mathcal P_{t,2}^2
       (1+\ii\tilde\nu)(2+\ii\tilde\nu)(3+2\ii\tilde\nu).
 \label{eq:top-angular-closed}
\end{align}

The remaining time coefficients can also be substituted explicitly.
Using \eqref{eq:top-finite-master}, set
\begin{align}
 \mathcal Z_t&=\frac{2\mathcal F_t}{1+2\ii\tilde\nu}
 -(1+2\ii\tilde\nu)\mathcal E_t
 +\frac{\ii\tilde\nu(5+2\ii\tilde\nu)}
 {2(1+\ii\tilde\nu)(2+\ii\tilde\nu)},\nonumber\\
 \mathcal N_t&=\mathcal B_t\Gamma^2(1+2\ii\tilde\nu)
 \cosh^2(\pi\tilde\nu)\nonumber\\
 &=\frac{2^{4\ii\tilde\nu}\pi\Gamma(-\tfrac12-2\ii\tilde\nu)
 \Gamma^4(1+\ii\tilde\nu)}
 {(4\pi)^{7/2}\Gamma(2+2\ii\tilde\nu)
 (5+2\ii\tilde\nu)(3+2\ii\tilde\nu)
 (5+4\ii\tilde\nu)(3+4\ii\tilde\nu)}.
 \label{eq:top-angular-explicit-common}
\end{align}
Gamma duplication and reflection give the second form of
$\mathcal N_t$. The five coefficients then become
\begin{align}
 A_{1,t}={}&-\frac{2(1+\ii\tilde\nu)}{1+2\ii\tilde\nu}\mathcal N_t
 \Bigg[(7-9\ii\tilde\nu-4\tilde\nu^2-4\ii\tilde\nu^3)\mathcal Z_t^2
 \nonumber\\
 &\hspace{18mm}
 +\frac{8(1+\tilde\nu^2)(5+2\ii\tilde\nu)^2}{2+\ii\tilde\nu}\mathcal Z_t
 +\frac{12(1+\ii\tilde\nu)^2(5+2\ii\tilde\nu)^3}{(2+\ii\tilde\nu)^2}
 \Bigg],\nonumber\\
 A_{2,t}={}&2(1+\ii\tilde\nu)^2(1+2\ii\tilde\nu)\mathcal N_t\mathcal Z_t^2,
 \qquad A_{3,t}=-3A_{2,t},\nonumber\\
 B_t={}&(1+\ii\tilde\nu)(1+2\ii\tilde\nu)(3+2\ii\tilde\nu)
 \mathcal N_t\mathcal Z_t^2,\nonumber\\
 D_t={}&-2(1+\ii\tilde\nu)(2+\ii\tilde\nu)(3+2\ii\tilde\nu)
 \mathcal N_t\mathcal Z_t^2.
 \label{eq:top-angular-explicit-finite}
\end{align}
Thus all five functions depend on the single hypergeometric function
$\mathcal F_t$, together with Gamma and elementary functions.
